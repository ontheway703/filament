# Filament Android 后端驱动实现

## 概述

Filament 采用多后端架构，在 Android 平台支持 OpenGL ES、Vulkan 和实验性的 WebGPU 后端。本文档深入分析各个后端的实现原理、性能特性和使用场景。

## 后端架构总览

### 架构层次
```
应用层 (Java/Kotlin)
    ↓
JNI 绑定层
    ↓
Filament 引擎层 (C++)
    ↓
后端抽象层 (Backend Interface)
    ↓
具体后端实现
├── OpenGL ES Backend
├── Vulkan Backend
└── WebGPU Backend (实验性)
```

### 后端选择机制
```cpp
// filament/src/Engine.cpp
Engine* Engine::create(Backend backend) {
    switch (backend) {
        case Backend::OPENGL:
            return createOpenGLEngine();
        case Backend::VULKAN:
            return createVulkanEngine();
        case Backend::WEBGPU:
            return createWebGPUEngine();
        case Backend::DEFAULT:
            return createDefaultEngine(); // 自动选择最佳后端
    }
}

Engine* Engine::createDefaultEngine() {
    // Android 上的后端选择优先级
    if (isVulkanSupported()) {
        return createVulkanEngine();
    } else if (isOpenGLESSupported()) {
        return createOpenGLEngine();
    } else {
        // 降级到软件渲染
        return createSoftwareEngine();
    }
}
```

```cpp
class OpenGLRenderState {
    struct {
        GLenum cullFace = GL_BACK;
        GLenum frontFace = GL_CCW;
        bool cullingEnabled = false;
    } mCulling;

    struct {
        GLenum func = GL_LESS;
        bool testEnabled = false;
        bool writeEnabled = true;
    } mDepth;

    struct {
        GLenum sfactor = GL_ONE;
        GLenum dfactor = GL_ZERO;
        bool blendEnabled = false;
    } mBlending;

public:
    void setCulling(bool enabled, GLenum face, GLenum front) {
        if (mCulling.cullingEnabled != enabled) {
            if (enabled) {
                glEnable(GL_CULL_FACE);
            } else {
                glDisable(GL_CULL_FACE);
            }
            mCulling.cullingEnabled = enabled;
        }

        if (enabled) {
            if (mCulling.cullFace != face) {
                glCullFace(face);
                mCulling.cullFace = face;
            }
            if (mCulling.frontFace != front) {
                glFrontFace(front);
                mCulling.frontFace = front;
            }
        }
    }
};
```

```cpp
// OpenGL 纹理对象
struct OpenGLTexture {
    GLuint textureId = 0;
    GLenum target = GL_TEXTURE_2D;
    GLsizei width = 0;
    GLsizei height = 0;
    GLsizei depth = 0;
    GLenum internalFormat = GL_RGBA8;
    GLsizei levels = 1;

    // Android 特定: EGL 图像支持
    EGLImageKHR eglImage = EGL_NO_IMAGE_KHR;
    bool isExternalTexture = false;
};

Handle<HwTexture> OpenGLDriver::createTexture(
    SamplerType target, uint8_t levels,
    TextureFormat format, uint8_t samples,
    uint32_t width, uint32_t height, uint32_t depth,
    TextureUsage usage) {

    OpenGLTexture* texture = new OpenGLTexture();

    // 选择 OpenGL 目标
    switch (target) {
        case SamplerType::SAMPLER_2D:
            texture->target = GL_TEXTURE_2D;
            break;
        case SamplerType::SAMPLER_CUBEMAP:
            texture->target = GL_TEXTURE_CUBE_MAP;
            break;
        case SamplerType::SAMPLER_EXTERNAL:
            texture->target = GL_TEXTURE_EXTERNAL_OES;
            texture->isExternalTexture = true;
            break;
    }

    // 生成纹理对象
    glGenTextures(1, &texture->textureId);
    glBindTexture(texture->target, texture->textureId);

    // 设置纹理参数
    setupTextureParameters(texture, target, levels, format, samples);

    // 分配存储空间
    if (!texture->isExternalTexture) {
        allocateTextureStorage(texture, levels, format, width, height, depth);
    }

    return texture;
}
```

```cpp
class OpenGLMemoryManager {
private:
    size_t mTotalGPUMemory = 0;
    size_t mUsedGPUMemory = 0;
    std::unordered_map<GLuint, size_t> mTextureMemory;
    std::unordered_map<GLuint, size_t> mBufferMemory;

public:
    void trackTextureMemory(GLuint textureId, size_t size) {
        mTextureMemory[textureId] = size;
        mUsedGPUMemory += size;

        if (mUsedGPUMemory > mTotalGPUMemory * 0.8f) {
            // 触发垃圾回收
            triggerMemoryCleanup();
        }
    }

    void releaseTextureMemory(GLuint textureId) {
        auto it = mTextureMemory.find(textureId);
        if (it != mTextureMemory.end()) {
            mUsedGPUMemory -= it->second;
            mTextureMemory.erase(it);
        }
    }
};
```

#### 渲染管线
```cpp
class VulkanRenderPipeline {
private:
    VkPipeline mGraphicsPipeline;
    VkPipelineLayout mPipelineLayout;
    VkRenderPass mRenderPass;

public:
    void createGraphicsPipeline(const PipelineState& state) {
        // 顶点输入状态
        VkPipelineVertexInputStateCreateInfo vertexInputInfo = {};
        vertexInputInfo.sType = VK_STRUCTURE_TYPE_PIPELINE_VERTEX_INPUT_STATE_CREATE_INFO;
        setupVertexInput(vertexInputInfo, state.vertexLayout);

        // 输入装配状态
        VkPipelineInputAssemblyStateCreateInfo inputAssembly = {};
        inputAssembly.sType = VK_STRUCTURE_TYPE_PIPELINE_INPUT_ASSEMBLY_STATE_CREATE_INFO;
        inputAssembly.topology = getPrimitiveTopology(state.primitiveType);
        inputAssembly.primitiveRestartEnable = VK_FALSE;

        // 视口状态
        VkPipelineViewportStateCreateInfo viewportState = {};
        viewportState.sType = VK_STRUCTURE_TYPE_PIPELINE_VIEWPORT_STATE_CREATE_INFO;
        viewportState.viewportCount = 1;
        viewportState.scissorCount = 1;

        // 光栅化状态
        VkPipelineRasterizationStateCreateInfo rasterizer = {};
        rasterizer.sType = VK_STRUCTURE_TYPE_PIPELINE_RASTERIZATION_STATE_CREATE_INFO;
        rasterizer.depthClampEnable = VK_FALSE;
        rasterizer.rasterizerDiscardEnable = VK_FALSE;
        rasterizer.polygonMode = VK_POLYGON_MODE_FILL;
        rasterizer.lineWidth = 1.0f;
        rasterizer.cullMode = getCullMode(state.culling);
        rasterizer.frontFace = getFrontFace(state.frontFace);

        // 多重采样状态
        VkPipelineMultisampleStateCreateInfo multisampling = {};
        multisampling.sType = VK_STRUCTURE_TYPE_PIPELINE_MULTISAMPLE_STATE_CREATE_INFO;
        multisampling.sampleShadingEnable = VK_FALSE;
        multisampling.rasterizationSamples = getSampleCount(state.samples);

        // 深度模板状态
        VkPipelineDepthStencilStateCreateInfo depthStencil = {};
        depthStencil.sType = VK_STRUCTURE_TYPE_PIPELINE_DEPTH_STENCIL_STATE_CREATE_INFO;
        setupDepthStencil(depthStencil, state.depthState, state.stencilState);

        // 颜色混合状态
        VkPipelineColorBlendStateCreateInfo colorBlending = {};
        colorBlending.sType = VK_STRUCTURE_TYPE_PIPELINE_COLOR_BLEND_STATE_CREATE_INFO;
        setupColorBlending(colorBlending, state.blendState);

        // 创建图形管线
        VkGraphicsPipelineCreateInfo pipelineInfo = {};
        pipelineInfo.sType = VK_STRUCTURE_TYPE_GRAPHICS_PIPELINE_CREATE_INFO;
        pipelineInfo.stageCount = 2; // 顶点 + 片段着色器
        pipelineInfo.pStages = shaderStages;
        pipelineInfo.pVertexInputState = &vertexInputInfo;
        pipelineInfo.pInputAssemblyState = &inputAssembly;
        pipelineInfo.pViewportState = &viewportState;
        pipelineInfo.pRasterizationState = &rasterizer;
        pipelineInfo.pMultisampleState = &multisampling;
        pipelineInfo.pDepthStencilState = &depthStencil;
        pipelineInfo.pColorBlendState = &colorBlending;
        pipelineInfo.layout = mPipelineLayout;
        pipelineInfo.renderPass = mRenderPass;
        pipelineInfo.subpass = 0;

        VkResult result = vkCreateGraphicsPipelines(mDevice, VK_NULL_HANDLE, 1,
                                                   &pipelineInfo, nullptr, &mGraphicsPipeline);
        assert(result == VK_SUCCESS);
    }
};
```

```cpp
class BackendSelector {
public:
    static Backend selectOptimalBackend() {
        // 检查 Vulkan 支持
        if (isVulkanSupported()) {
            // 检查设备性能等级
            if (getPerformanceTier() >= PerformanceTier::HIGH) {
                return Backend::VULKAN;
            }

            // 检查复杂度需求
            if (getSceneComplexity() >= SceneComplexity::HIGH) {
                return Backend::VULKAN;
            }
        }

        // 默认使用 OpenGL ES
        return Backend::OPENGL;
    }

private:
    static bool isVulkanSupported() {
        // 检查 Android 版本
        if (android_get_device_api_level() < 24) {
            return false;
        }

        // 检查驱动支持
        return checkVulkanDriverSupport();
    }

    static PerformanceTier getPerformanceTier() {
        // 基于 CPU/GPU 型号判断性能等级
        std::string chipset = getChipsetModel();

        if (chipset.find("888") != std::string::npos ||
            chipset.find("8 Gen") != std::string::npos) {
            return PerformanceTier::HIGH;
        } else if (chipset.find("855") != std::string::npos ||
                   chipset.find("865") != std::string::npos) {
            return PerformanceTier::MEDIUM;
        } else {
            return PerformanceTier::LOW;
        }
    }
};
```

```cpp
class PerformanceMonitor {
private:
    std::chrono::high_resolution_clock::time_point mFrameStart;
    std::vector<float> mFrameTimes;
    size_t mFrameIndex = 0;

    // GPU 时间查询 (OpenGL)
    GLuint mTimeQueries[2];
    bool mQuerySupported = false;

public:
    void beginFrame() {
        mFrameStart = std::chrono::high_resolution_clock::now();

        if (mQuerySupported) {
            glBeginQuery(GL_TIME_ELAPSED, mTimeQueries[mFrameIndex % 2]);
        }
    }

    void endFrame() {
        if (mQuerySupported) {
            glEndQuery(GL_TIME_ELAPSED);
        }

        auto frameEnd = std::chrono::high_resolution_clock::now();
        float frameTime = std::chrono::duration<float, std::milli>(
            frameEnd - mFrameStart).count();

        mFrameTimes[mFrameIndex % mFrameTimes.size()] = frameTime;
        mFrameIndex++;

        // 每秒输出性能统计
        if (mFrameIndex % 60 == 0) {
            outputPerformanceStats();
        }
    }

    void outputPerformanceStats() {
        float avgFrameTime = 0.0f;
        for (float time : mFrameTimes) {
            avgFrameTime += time;
        }
        avgFrameTime /= mFrameTimes.size();

        float fps = 1000.0f / avgFrameTime;

        utils::slog.i << "Performance - FPS: " << fps
                     << ", Frame Time: " << avgFrameTime << "ms"
                     << utils::io::endl;

        // GPU 时间查询结果
        if (mQuerySupported) {
            GLuint64 gpuTime;
            glGetQueryObjectui64v(mTimeQueries[(mFrameIndex + 1) % 2],
                                 GL_QUERY_RESULT, &gpuTime);
            float gpuTimeMs = gpuTime / 1000000.0f;

            utils::slog.i << "GPU Time: " << gpuTimeMs << "ms" << utils::io::endl;
        }
    }
};
```

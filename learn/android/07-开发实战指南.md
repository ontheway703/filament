```java
public class FilamentManager {
    private static FilamentManager sInstance;
    private Engine mEngine;
    private Renderer mRenderer;
    private Scene mScene;
    private View mView;
    private Camera mCamera;
    private SwapChain mSwapChain;

    private final Object mLock = new Object();
    private boolean mInitialized = false;

    public static FilamentManager getInstance() {
        if (sInstance == null) {
            synchronized (FilamentManager.class) {
                if (sInstance == null) {
                    sInstance = new FilamentManager();
                }
            }
        }
        return sInstance;
    }

    public boolean initialize(Surface surface) {
        synchronized (mLock) {
            if (mInitialized) {
                return true;
            }

            try {
                // 创建引擎 - 自动选择最佳后端
                mEngine = Engine.create();

                // 创建渲染器
                mRenderer = mEngine.createRenderer();

                // 创建场景
                mScene = mEngine.createScene();

                // 创建视图
                mView = mEngine.createView();
                mView.setScene(mScene);

                // 创建摄像机
                int cameraEntity = EntityManager.get().create();
                mCamera = mEngine.createCamera(cameraEntity);
                mView.setCamera(mCamera);

                // 创建交换链
                if (surface != null) {
                    mSwapChain = mEngine.createSwapChain(surface);
                }

                // 设置默认渲染质量
                configureRenderQuality();

                // 设置默认光照
                setupDefaultLighting();

                mInitialized = true;
                return true;

            } catch (Exception e) {
                Log.e("FilamentManager", "Failed to initialize Filament", e);
                cleanup();
                return false;
            }
        }
    }

    private void configureRenderQuality() {
        // 根据设备性能调整渲染质量
        String deviceModel = Build.MODEL.toLowerCase();
        boolean isHighEndDevice = isHighEndDevice(deviceModel);

        View.RenderQuality.Builder qualityBuilder = new View.RenderQuality.Builder();

        if (isHighEndDevice) {
            qualityBuilder.hdrColorBuffer(View.QualityLevel.HIGH)
                         .msaa(View.QualityLevel.HIGH)
                         .dithering(View.Dithering.TEMPORAL);
        } else {
            qualityBuilder.hdrColorBuffer(View.QualityLevel.MEDIUM)
                         .msaa(View.QualityLevel.LOW)
                         .dithering(View.Dithering.NONE);
        }

        mView.setRenderQuality(qualityBuilder.build());
    }

    private boolean isHighEndDevice(String deviceModel) {
        // 基于设备型号判断性能等级
        return deviceModel.contains("pixel") &&
               (deviceModel.contains("6") || deviceModel.contains("7") || deviceModel.contains("8")) ||
               deviceModel.contains("galaxy s2") ||
               deviceModel.contains("oneplus");
    }

    public void render() {
        synchronized (mLock) {
            if (!mInitialized || mSwapChain == null) {
                return;
            }

            if (mRenderer.beginFrame(mSwapChain, 0)) {
                mRenderer.render(mView);
                mRenderer.endFrame();
            }
        }
    }

    public void cleanup() {
        synchronized (mLock) {
            if (mEngine != null) {
                if (mSwapChain != null) {
                    mEngine.destroySwapChain(mSwapChain);
                    mSwapChain = null;
                }
                if (mRenderer != null) {
                    mEngine.destroyRenderer(mRenderer);
                    mRenderer = null;
                }
                if (mView != null) {
                    mEngine.destroyView(mView);
                    mView = null;
                }
                if (mScene != null) {
                    mEngine.destroyScene(mScene);
                    mScene = null;
                }
                if (mCamera != null) {
                    mEngine.destroyCamera(mCamera);
                    mCamera = null;
                }

                mEngine.destroy();
                mEngine = null;
            }
            mInitialized = false;
        }
    }
}
```

### 2. 资源管理系统

#### ResourceManager 实现
```java
public class ResourceManager {
    private static final String TAG = "ResourceManager";

    private final Engine mEngine;
    private final AssetManager mAssetManager;

    // 资源缓存
    private final Map<String, Material> mMaterialCache = new ConcurrentHashMap<>();
    private final Map<String, Texture> mTextureCache = new ConcurrentHashMap<>();
    private final Map<String, FilamentAsset> mModelCache = new ConcurrentHashMap<>();

    // 资源加载状态
    private final Map<String, CompletableFuture<Material>> mMaterialLoadingTasks = new ConcurrentHashMap<>();
    private final Map<String, CompletableFuture<Texture>> mTextureLoadingTasks = new ConcurrentHashMap<>();

    public ResourceManager(Engine engine, AssetManager assetManager) {
        mEngine = engine;
        mAssetManager = assetManager;
    }

    public CompletableFuture<Material> loadMaterial(String path) {
        // 检查缓存
        Material cached = mMaterialCache.get(path);
        if (cached != null) {
            return CompletableFuture.completedFuture(cached);
        }

        // 检查是否正在加载
        CompletableFuture<Material> loadingTask = mMaterialLoadingTasks.get(path);
        if (loadingTask != null) {
            return loadingTask;
        }

        // 开始异步加载
        CompletableFuture<Material> future = CompletableFuture.supplyAsync(() -> {
            try {
                ByteBuffer buffer = loadAssetBuffer(path);
                Material material = new Material.Builder()
                    .payload(buffer, buffer.remaining())
                    .build(mEngine);

                mMaterialCache.put(path, material);
                return material;

            } catch (Exception e) {
                Log.e(TAG, "Failed to load material: " + path, e);
                throw new RuntimeException(e);
            }
        }).whenComplete((result, throwable) -> {
            mMaterialLoadingTasks.remove(path);
        });

        mMaterialLoadingTasks.put(path, future);
        return future;
    }

    public CompletableFuture<Texture> loadTexture(String path, boolean sRGB) {
        String cacheKey = path + "_" + sRGB;

        Texture cached = mTextureCache.get(cacheKey);
        if (cached != null) {
            return CompletableFuture.completedFuture(cached);
        }

        CompletableFuture<Texture> loadingTask = mTextureLoadingTasks.get(cacheKey);
        if (loadingTask != null) {
            return loadingTask;
        }

        CompletableFuture<Texture> future = CompletableFuture.supplyAsync(() -> {
            try {
                Bitmap bitmap = loadBitmapFromAssets(path);

                Texture.InternalFormat format = sRGB ?
                    Texture.InternalFormat.SRGB8_A8 :
                    Texture.InternalFormat.RGBA8;

                Texture texture = new Texture.Builder()
                    .width(bitmap.getWidth())
                    .height(bitmap.getHeight())
                    .levels(calculateMipLevels(bitmap.getWidth(), bitmap.getHeight()))
                    .format(format)
                    .build(mEngine);

                // 设置纹理数据
                Texture.PixelBufferDescriptor buffer = new Texture.PixelBufferDescriptor(
                    bitmapToByteBuffer(bitmap),
                    Texture.Format.RGBA,
                    Texture.Type.UBYTE);

                texture.setImage(mEngine, 0, buffer);
                texture.generateMipmaps(mEngine);

                mTextureCache.put(cacheKey, texture);
                return texture;

            } catch (Exception e) {
                Log.e(TAG, "Failed to load texture: " + path, e);
                throw new RuntimeException(e);
            }
        }).whenComplete((result, throwable) -> {
            mTextureLoadingTasks.remove(cacheKey);
        });

        mTextureLoadingTasks.put(cacheKey, future);
        return future;
    }

    private ByteBuffer loadAssetBuffer(String path) throws IOException {
        InputStream inputStream = mAssetManager.open(path);
        byte[] bytes = new byte[inputStream.available()];
        inputStream.read(bytes);
        inputStream.close();

        ByteBuffer buffer = ByteBuffer.allocateDirect(bytes.length);
        buffer.put(bytes);
        buffer.rewind();
        return buffer;
    }

    private Bitmap loadBitmapFromAssets(String path) throws IOException {
        InputStream inputStream = mAssetManager.open(path);
        Bitmap bitmap = BitmapFactory.decodeStream(inputStream);
        inputStream.close();
        return bitmap;
    }

    private int calculateMipLevels(int width, int height) {
        return (int) Math.floor(Math.log(Math.max(width, height)) / Math.log(2)) + 1;
    }

    public void cleanup() {
        // 清理所有缓存的资源
        for (Material material : mMaterialCache.values()) {
            mEngine.destroyMaterial(material);
        }
        mMaterialCache.clear();

        for (Texture texture : mTextureCache.values()) {
            mEngine.destroyTexture(texture);
        }
        mTextureCache.clear();

        for (FilamentAsset asset : mModelCache.values()) {
            asset.releaseSourceData();
        }
        mModelCache.clear();
    }
}
```

```java
public class SceneManager {
    private final Engine mEngine;
    private final Scene mScene;
    private final ResourceManager mResourceManager;

    // 场景对象管理
    private final Map<String, SceneObject> mSceneObjects = new ConcurrentHashMap<>();
    private final Map<String, LightSource> mLights = new ConcurrentHashMap<>();

    // 动画系统
    private final List<Animator> mAnimators = new ArrayList<>();

    public SceneManager(Engine engine, Scene scene, ResourceManager resourceManager) {
        mEngine = engine;
        mScene = scene;
        mResourceManager = resourceManager;

        setupDefaultEnvironment();
    }

    private void setupDefaultEnvironment() {
        // 设置默认的 IBL 环境
        loadEnvironmentAsync("environments/studio.ktx");

        // 添加主光源
        addDirectionalLight("sun",
            new float[]{0.0f, -1.0f, -1.0f},  // 方向
            new float[]{1.0f, 1.0f, 0.9f},    // 颜色
            100000.0f);                        // 强度
    }

    public CompletableFuture<SceneObject> loadModel(String name, String modelPath) {
        return CompletableFuture.supplyAsync(() -> {
            try {
                // 使用 gltfio 加载模型
                AssetLoader assetLoader = new AssetLoader(mEngine,
                    new UbershaderProvider(mEngine), EntityManager.get());

                ByteBuffer buffer = mResourceManager.loadAssetBuffer(modelPath);
                FilamentAsset asset = assetLoader.createAsset(buffer);

                if (asset == null) {
                    throw new RuntimeException("Failed to load model: " + modelPath);
                }

                // 释放源数据，节省内存
                asset.releaseSourceData();

                // 添加到场景
                mScene.addEntities(asset.getEntities());

                // 创建场景对象
                SceneObject sceneObject = new SceneObject(name, asset, mEngine);
                mSceneObjects.put(name, sceneObject);

                // 设置动画器
                if (asset.getAnimator() != null) {
                    mAnimators.add(asset.getAnimator());
                }

                return sceneObject;

            } catch (Exception e) {
                Log.e("SceneManager", "Failed to load model: " + modelPath, e);
                throw new RuntimeException(e);
            }
        });
    }

    public void addDirectionalLight(String name, float[] direction, float[] color, float intensity) {
        int entity = EntityManager.get().create();

        LightManager.Builder(LightManager.Type.DIRECTIONAL)
            .color(color[0], color[1], color[2])
            .intensity(intensity)
            .direction(direction[0], direction[1], direction[2])
            .castShadows(true)
            .build(mEngine, entity);

        mScene.addEntity(entity);

        LightSource light = new LightSource(name, entity, LightManager.Type.DIRECTIONAL);
        mLights.put(name, light);
    }

    public void updateAnimations(float deltaTime) {
        for (Animator animator : mAnimators) {
            // 更新所有动画
            for (int i = 0; i < animator.getAnimationCount(); i++) {
                animator.applyAnimation(i, getCurrentAnimationTime(i));
            }
            animator.updateBoneMatrices();
        }
    }

    private float getCurrentAnimationTime(int animationIndex) {
        // 基于系统时间计算动画时间
        long currentTime = System.nanoTime();
        return (currentTime % (long)(1e9 * 10)) / 1e9f; // 10秒循环
    }

    public SceneObject getSceneObject(String name) {
        return mSceneObjects.get(name);
    }

    public void removeSceneObject(String name) {
        SceneObject object = mSceneObjects.remove(name);
        if (object != null) {
            mScene.removeEntities(object.getAsset().getEntities());
            object.cleanup();
        }
    }

    public void cleanup() {
        for (SceneObject object : mSceneObjects.values()) {
            mScene.removeEntities(object.getAsset().getEntities());
            object.cleanup();
        }
        mSceneObjects.clear();

        for (LightSource light : mLights.values()) {
            mScene.removeEntity(light.getEntity());
            EntityManager.get().destroy(light.getEntity());
        }
        mLights.clear();

        mAnimators.clear();
    }
}

// 场景对象封装类
public class SceneObject {
    private final String mName;
    private final FilamentAsset mAsset;
    private final Engine mEngine;
    private final Matrix4f mTransform = new Matrix4f();

    public SceneObject(String name, FilamentAsset asset, Engine engine) {
        mName = name;
        mAsset = asset;
        mEngine = engine;
        mTransform.identity();
    }

    public void setTransform(Matrix4f transform) {
        mTransform.set(transform);

        // 应用变换到所有实体
        int[] entities = mAsset.getEntities();
        TransformManager tm = mEngine.getTransformManager();

        for (int entity : entities) {
            if (tm.hasComponent(entity)) {
                TransformManager.Instance instance = tm.getInstance(entity);
                tm.setTransform(instance,
                    mTransform.m00(), mTransform.m10(), mTransform.m20(), mTransform.m30(),
                    mTransform.m01(), mTransform.m11(), mTransform.m21(), mTransform.m31(),
                    mTransform.m02(), mTransform.m12(), mTransform.m22(), mTransform.m32());
            }
        }
    }

    public FilamentAsset getAsset() {
        return mAsset;
    }

    public void cleanup() {
        if (mAsset != null) {
            mAsset.releaseSourceData();
        }
    }
}
```

```java
public class LODManager {
    private final Camera mCamera;
    private final Map<String, LODObject> mLODObjects = new ConcurrentHashMap<>();

    public class LODObject {
        private final List<SceneObject> mLODLevels;
        private final List<Float> mDistanceThresholds;
        private int mCurrentLOD = 0;

        public LODObject(List<SceneObject> lodLevels, List<Float> thresholds) {
            mLODLevels = lodLevels;
            mDistanceThresholds = thresholds;
        }

        public void updateLOD(float distanceToCamera) {
            int newLOD = 0;
            for (int i = 0; i < mDistanceThresholds.size(); i++) {
                if (distanceToCamera > mDistanceThresholds.get(i)) {
                    newLOD = i + 1;
                } else {
                    break;
                }
            }

            if (newLOD != mCurrentLOD) {
                // 隐藏当前LOD
                setLODVisibility(mCurrentLOD, false);

                // 显示新LOD
                mCurrentLOD = Math.min(newLOD, mLODLevels.size() - 1);
                setLODVisibility(mCurrentLOD, true);
            }
        }

        private void setLODVisibility(int lodIndex, boolean visible) {
            if (lodIndex < mLODLevels.size()) {
                SceneObject object = mLODLevels.get(lodIndex);
                // 设置对象可见性
                setObjectVisibility(object, visible);
            }
        }
    }

    public void updateAllLODs() {
        float[] cameraPosition = getCameraPosition();

        for (LODObject lodObject : mLODObjects.values()) {
            float distance = calculateDistanceToCamera(lodObject, cameraPosition);
            lodObject.updateLOD(distance);
        }
    }
}
```

#### 纹理流式加载
```java
public class TextureStreamingManager {
    private final Engine mEngine;
    private final int mMaxTextureMemory = 256 * 1024 * 1024; // 256MB
    private int mCurrentTextureMemory = 0;

    private final LRUCache<String, TextureEntry> mTextureCache;

    public TextureStreamingManager(Engine engine) {
        mEngine = engine;
        mTextureCache = new LRUCache<String, TextureEntry>(100) {
            @Override
            protected void entryRemoved(boolean evicted, String key,
                                       TextureEntry oldValue, TextureEntry newValue) {
                if (evicted) {
                    unloadTexture(oldValue);
                }
            }
        };
    }

    public CompletableFuture<Texture> loadTexture(String path, int priority) {
        TextureEntry entry = mTextureCache.get(path);
        if (entry != null && entry.texture != null) {
            entry.lastAccessed = System.currentTimeMillis();
            entry.priority = Math.max(entry.priority, priority);
            return CompletableFuture.completedFuture(entry.texture);
        }

        return CompletableFuture.supplyAsync(() -> {
            // 检查内存限制
            checkMemoryLimit();

            try {
                Texture texture = loadTextureFromPath(path);
                int textureSize = calculateTextureSize(texture);

                TextureEntry newEntry = new TextureEntry();
                newEntry.texture = texture;
                newEntry.size = textureSize;
                newEntry.priority = priority;
                newEntry.lastAccessed = System.currentTimeMillis();

                mTextureCache.put(path, newEntry);
                mCurrentTextureMemory += textureSize;

                return texture;

            } catch (Exception e) {
                throw new RuntimeException("Failed to load texture: " + path, e);
            }
        });
    }

    private void checkMemoryLimit() {
        while (mCurrentTextureMemory > mMaxTextureMemory && mTextureCache.size() > 0) {
            // 找到优先级最低且最少使用的纹理
            String keyToRemove = findLeastImportantTexture();
            if (keyToRemove != null) {
                TextureEntry entry = mTextureCache.remove(keyToRemove);
                unloadTexture(entry);
            }
        }
    }

    private void unloadTexture(TextureEntry entry) {
        if (entry.texture != null) {
            mEngine.destroyTexture(entry.texture);
            mCurrentTextureMemory -= entry.size;
        }
    }

    private static class TextureEntry {
        Texture texture;
        int size;
        int priority;
        long lastAccessed;
    }
}
```

#### 自适应帧率控制
```java
public class AdaptiveFrameRateController {
    private static final int TARGET_FPS_HIGH = 60;
    private static final int TARGET_FPS_MEDIUM = 30;
    private static final int TARGET_FPS_LOW = 20;

    private final BatteryManager mBatteryManager;
    private final ThermalManager mThermalManager;

    private int mCurrentTargetFPS = TARGET_FPS_HIGH;
    private long mLastFrameTime = 0;
    private boolean mIsCharging = false;
    private float mBatteryLevel = 1.0f;

    public AdaptiveFrameRateController(Context context) {
        mBatteryManager = (BatteryManager) context.getSystemService(Context.BATTERY_SERVICE);

        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {
            mThermalManager = (ThermalManager) context.getSystemService(Context.THERMAL_SERVICE);
        }

        // 监听电池状态变化
        registerBatteryReceiver(context);
    }

    public boolean shouldRenderFrame() {
        long currentTime = System.nanoTime();
        long targetFrameTime = 1_000_000_000L / mCurrentTargetFPS;

        if (currentTime - mLastFrameTime >= targetFrameTime) {
            mLastFrameTime = currentTime;
            return true;
        }

        return false;
    }

    public void updateFrameRateTarget() {
        updateBatteryInfo();

        int newTargetFPS = calculateOptimalFPS();

        if (newTargetFPS != mCurrentTargetFPS) {
            mCurrentTargetFPS = newTargetFPS;
            Log.i("FrameRate", "Target FPS changed to: " + mCurrentTargetFPS);

            // 调整渲染质量
            adjustRenderQuality();
        }
    }

    private int calculateOptimalFPS() {
        // 充电状态下使用高帧率
        if (mIsCharging) {
            return TARGET_FPS_HIGH;
        }

        // 低电量时降低帧率
        if (mBatteryLevel < 0.15f) {
            return TARGET_FPS_LOW;
        } else if (mBatteryLevel < 0.30f) {
            return TARGET_FPS_MEDIUM;
        }

        // 检查热节流
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {
            int thermalStatus = mThermalManager.getCurrentThermalStatus();
            if (thermalStatus >= ThermalManager.THERMAL_STATUS_MODERATE) {
                return TARGET_FPS_MEDIUM;
            }
            if (thermalStatus >= ThermalManager.THERMAL_STATUS_SEVERE) {
                return TARGET_FPS_LOW;
            }
        }

        return TARGET_FPS_HIGH;
    }

    private void adjustRenderQuality() {
        // 根据目标帧率调整渲染质量
        // 这需要与 FilamentManager 配合
        FilamentManager.getInstance().adjustQualityForFrameRate(mCurrentTargetFPS);
    }
}
```

### 1. 内存泄漏问题

#### 问题排查
```java
public class FilamentMemoryMonitor {
    private static final String TAG = "FilamentMemory";

    private final Runtime mRuntime;
    private long mLastNativeMemory = 0;
    private long mLastJavaMemory = 0;

    public FilamentMemoryMonitor() {
        mRuntime = Runtime.getRuntime();
    }

    public void logMemoryUsage(String tag) {
        // Java 堆内存
        long javaTotal = mRuntime.totalMemory();
        long javaFree = mRuntime.freeMemory();
        long javaUsed = javaTotal - javaFree;

        // 原生内存 (估算)
        Debug.MemoryInfo memInfo = new Debug.MemoryInfo();
        Debug.getMemoryInfo(memInfo);
        long nativeMemory = memInfo.nativePss * 1024;

        Log.i(TAG, String.format("%s - Java: %dMB, Native: %dMB",
            tag, javaUsed / (1024 * 1024), nativeMemory / (1024 * 1024)));

        // 检查内存增长
        checkMemoryGrowth(javaUsed, nativeMemory);

        mLastJavaMemory = javaUsed;
        mLastNativeMemory = nativeMemory;
    }

    private void checkMemoryGrowth(long javaUsed, long nativeMemory) {
        if (mLastJavaMemory > 0) {
            long javaGrowth = javaUsed - mLastJavaMemory;
            long nativeGrowth = nativeMemory - mLastNativeMemory;

            if (javaGrowth > 10 * 1024 * 1024) { // 10MB
                Log.w(TAG, "Large Java memory growth: " + (javaGrowth / (1024 * 1024)) + "MB");
            }

            if (nativeGrowth > 20 * 1024 * 1024) { // 20MB
                Log.w(TAG, "Large native memory growth: " + (nativeGrowth / (1024 * 1024)) + "MB");
            }
        }
    }

    public void dumpMemoryInfo() {
        System.gc(); // 强制垃圾回收

        try {
            Thread.sleep(100); // 等待GC完成
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
        }

        logMemoryUsage("After GC");

        // 检查是否有内存泄漏
        if (shouldTriggerMemoryDump()) {
            triggerMemoryDump();
        }
    }

    private boolean shouldTriggerMemoryDump() {
        long maxMemory = mRuntime.maxMemory();
        long usedMemory = mRuntime.totalMemory() - mRuntime.freeMemory();
        return (usedMemory * 100 / maxMemory) > 80; // 使用超过80%
    }

    private void triggerMemoryDump() {

    }
```

```java
        leaked.removeAll(mDestroyedResources);

        if (!leaked.isEmpty()) {
            Log.w("ResourceTracker", "Potential resource leaks detected:");
            for (String resource : leaked) {
                Log.w("ResourceTracker", "  - " + resource);
            }
        } else {
            Log.i("ResourceTracker", "No resource leaks detected");
        }
    }
```

```java
public class DeviceCompatibilityChecker {

    public static class CompatibilityInfo {
        public boolean filamentSupported;
        public boolean vulkanSupported;
        public boolean openglSupported;
        public String recommendedBackend;
        public List<String> issues;
        public PerformanceTier performanceTier;
    }

    public static CompatibilityInfo checkCompatibility(Context context) {
        CompatibilityInfo info = new CompatibilityInfo();
        info.issues = new ArrayList<>();

        // 检查 OpenGL ES 支持
        info.openglSupported = checkOpenGLSupport(context);
        if (!info.openglSupported) {
            info.issues.add("OpenGL ES 3.0+ not supported");
        }

        // 检查 Vulkan 支持
        info.vulkanSupported = checkVulkanSupport(context);

        // 检查 Filament 支持
        info.filamentSupported = checkFilamentSupport();
        if (!info.filamentSupported) {
            info.issues.add("Filament engine initialization failed");
        }

        // 推荐后端
        if (info.vulkanSupported && getPerformanceTier() >= PerformanceTier.MEDIUM) {
            info.recommendedBackend = "Vulkan";
        } else if (info.openglSupported) {
            info.recommendedBackend = "OpenGL ES";
        } else {
            info.recommendedBackend = "None (software fallback)";
        }

        info.performanceTier = getPerformanceTier();

        return info;
    }

    private static boolean checkOpenGLSupport(Context context) {
        ActivityManager am = (ActivityManager) context.getSystemService(Context.ACTIVITY_SERVICE);
        ConfigurationInfo configInfo = am.getDeviceConfigurationInfo();
        return configInfo.reqGlEsVersion >= 0x30000; // OpenGL ES 3.0
    }

    private static boolean checkVulkanSupport(Context context) {
        if (Build.VERSION.SDK_INT < Build.VERSION_CODES.N) {
            return false;
        }

        PackageManager pm = context.getPackageManager();
        return pm.hasSystemFeature(PackageManager.FEATURE_VULKAN_HARDWARE_LEVEL) &&
               pm.hasSystemFeature(PackageManager.FEATURE_VULKAN_HARDWARE_VERSION, 0x400003); // Vulkan 1.0.3
    }

    private static boolean checkFilamentSupport() {
        try {
            Engine engine = Engine.create();
            if (engine != null) {
                engine.destroy();
                return true;
            }
        } catch (Exception e) {
            Log.e("Compatibility", "Filament not supported", e);
        }
        return false;
    }

    public enum PerformanceTier {
        LOW, MEDIUM, HIGH
    }

    private static PerformanceTier getPerformanceTier() {
        // 基于设备信息判断性能等级
        String model = Build.MODEL.toLowerCase();
        String manufacturer = Build.MANUFACTURER.toLowerCase();

        // 高端设备
        if ((manufacturer.contains("samsung") && model.contains("galaxy s")) ||
            (manufacturer.contains("google") && model.contains("pixel")) ||
            manufacturer.contains("oneplus")) {
            return PerformanceTier.HIGH;
        }

        // 中端设备
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.N &&
            Runtime.getRuntime().maxMemory() > 2L * 1024 * 1024 * 1024) { // 2GB+
            return PerformanceTier.MEDIUM;
        }

        return PerformanceTier.LOW;
    }
}
```

```java
public class FilamentCrashReporter {
    private static final String TAG = "FilamentCrash";

    public static void init() {
        // 设置自定义崩溃处理器
        Thread.setDefaultUncaughtExceptionHandler(new FilamentUncaughtExceptionHandler());

        // 注册 Filament 原生崩溃处理
        registerNativeCrashHandler();
    }

    private static class FilamentUncaughtExceptionHandler implements Thread.UncaughtExceptionHandler {
        private final Thread.UncaughtExceptionHandler mDefaultHandler;

        public FilamentUncaughtExceptionHandler() {
            mDefaultHandler = Thread.getDefaultUncaughtExceptionHandler();
        }

        @Override
        public void uncaughtException(Thread thread, Throwable throwable) {
            try {
                // 收集 Filament 相关信息
                collectFilamentDebugInfo(throwable);

                // 上报到 Crashlytics
                FirebaseCrashlytics.getInstance().recordException(throwable);

            } catch (Exception e) {
                Log.e(TAG, "Error reporting crash", e);
            } finally {
                // 调用默认处理器
                if (mDefaultHandler != null) {
                    mDefaultHandler.uncaughtException(thread, throwable);
                }
            }
        }

        private void collectFilamentDebugInfo(Throwable throwable) {
            FirebaseCrashlytics crashlytics = FirebaseCrashlytics.getInstance();

            // 设备信息
            crashlytics.setCustomKey("device_model", Build.MODEL);
            crashlytics.setCustomKey("android_version", Build.VERSION.RELEASE);
            crashlytics.setCustomKey("api_level", Build.VERSION.SDK_INT);

            // Filament 信息
            try {
                crashlytics.setCustomKey("filament_version", Engine.getFilamentVersion());
                crashlytics.setCustomKey("backend_support_vulkan", Engine.isBackendSupported(Engine.Backend.VULKAN));
                crashlytics.setCustomKey("backend_support_opengl", Engine.isBackendSupported(Engine.Backend.OPENGL));
            } catch (Exception e) {
                crashlytics.setCustomKey("filament_init_error", e.getMessage());
            }

            // 内存信息
            Runtime runtime = Runtime.getRuntime();
            crashlytics.setCustomKey("memory_max", runtime.maxMemory() / (1024 * 1024));
            crashlytics.setCustomKey("memory_used", (runtime.totalMemory() - runtime.freeMemory()) / (1024 * 1024));

            // 堆栈信息
            if (throwable.getMessage() != null && throwable.getMessage().contains("filament")) {
                crashlytics.setCustomKey("filament_related", true);
            }
        }
    }

    private static native void registerNativeCrashHandler();
}
```

```java
public class FilamentRemoteConfig {
    private static final String TAG = "FilamentConfig";

    // 配置键
    private static final String KEY_ENABLE_VULKAN = "filament_enable_vulkan";
    private static final String KEY_MAX_TEXTURE_SIZE = "filament_max_texture_size";
    private static final String KEY_RENDER_QUALITY = "filament_render_quality";
    private static final String KEY_ENABLE_OCCLUSION_CULLING = "filament_enable_occlusion_culling";

    private final FirebaseRemoteConfig mRemoteConfig;
    private final SharedPreferences mPrefs;

    public FilamentRemoteConfig(Context context) {
        mRemoteConfig = FirebaseRemoteConfig.getInstance();
        mPrefs = context.getSharedPreferences("filament_config", Context.MODE_PRIVATE);

        setupDefaults();
        fetchConfig();
    }

    private void setupDefaults() {
        Map<String, Object> defaults = new HashMap<>();
        defaults.put(KEY_ENABLE_VULKAN, true);
        defaults.put(KEY_MAX_TEXTURE_SIZE, 2048);
        defaults.put(KEY_RENDER_QUALITY, "AUTO");
        defaults.put(KEY_ENABLE_OCCLUSION_CULLING, true);

        mRemoteConfig.setDefaultsAsync(defaults);
    }

    private void fetchConfig() {
        long cacheExpiration = 3600; // 1 hour in debug, 12 hours in release

        mRemoteConfig.fetch(cacheExpiration)
            .addOnCompleteListener(task -> {
                if (task.isSuccessful()) {
                    mRemoteConfig.activate();
                    Log.d(TAG, "Remote config fetched and activated");

                    // 应用新配置
                    applyConfig();
                } else {
                    Log.w(TAG, "Remote config fetch failed");
                }
            });
    }

    private void applyConfig() {
        // 缓存配置到本地
        SharedPreferences.Editor editor = mPrefs.edit();
        editor.putBoolean(KEY_ENABLE_VULKAN, mRemoteConfig.getBoolean(KEY_ENABLE_VULKAN));
        editor.putLong(KEY_MAX_TEXTURE_SIZE, mRemoteConfig.getLong(KEY_MAX_TEXTURE_SIZE));
        editor.putString(KEY_RENDER_QUALITY, mRemoteConfig.getString(KEY_RENDER_QUALITY));
        editor.putBoolean(KEY_ENABLE_OCCLUSION_CULLING, mRemoteConfig.getBoolean(KEY_ENABLE_OCCLUSION_CULLING));
        editor.apply();

        // 通知配置变化
        notifyConfigChanged();
    }

    public boolean isVulkanEnabled() {
        return mPrefs.getBoolean(KEY_ENABLE_VULKAN, true);
    }

    public int getMaxTextureSize() {
        return (int) mPrefs.getLong(KEY_MAX_TEXTURE_SIZE, 2048);
    }

    public String getRenderQuality() {
        return mPrefs.getString(KEY_RENDER_QUALITY, "AUTO");
    }

    public boolean isOcclusionCullingEnabled() {
        return mPrefs.getBoolean(KEY_ENABLE_OCCLUSION_CULLING, true);
    }
}
```

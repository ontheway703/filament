# Filament iOS 构建系统与工具链

## 构建系统概览

Filament iOS 使用基于 CMake 的跨平台构建系统，通过专门的 iOS 工具链文件和构建脚本实现自动化编译和打包。

```
构建流程:
桌面工具构建 → iOS 库编译 → 通用库创建 → Xcode 项目生成
    ↓              ↓           ↓             ↓
   matc,         静态库       Fat Binary    .xcodeproj
   resgen        (.a)         (lipo)        (XcodeGen)
```

## 核心构建工具

### 1. build.sh 脚本

**位置**: `./build.sh`
**功能**: 统一的构建入口脚本

#### 基础用法
```bash
# 构建桌面工具（必须先执行）
./build.sh -p desktop -i release

# 构建 iOS 库（仅设备）
./build.sh -p ios -i debug

# 构建 iOS 库（设备 + 模拟器）
./build.sh -s -p ios -i debug

# 安装到输出目录
./build.sh -p ios -i release
```

#### 重要参数
```bash
-p platform     # 平台: desktop, ios, android, webgl
-i install      # 安装到输出目录
-s              # 包含模拟器支持
-c              # 清理 CMake 缓存
-j N            # 并行编译任务数
```

### 2. CMake iOS 工具链

**位置**: `third_party/clang/iOS.cmake`
**功能**: iOS 平台的交叉编译配置

#### 工具链配置
```cmake
# iOS.cmake 核心配置
set(CMAKE_SYSTEM_NAME iOS)
set(CMAKE_SYSTEM_VERSION 11.0)

# 设置 iOS SDK 路径
execute_process(
    COMMAND xcrun --sdk iphoneos --show-sdk-path
    OUTPUT_VARIABLE CMAKE_OSX_SYSROOT
    OUTPUT_STRIP_TRAILING_WHITESPACE
)

# 编译器设置
set(CMAKE_C_COMPILER /usr/bin/clang)
set(CMAKE_CXX_COMPILER /usr/bin/clang++)

# 架构支持
if(IOS_ARCH STREQUAL "arm64")
    set(CMAKE_OSX_ARCHITECTURES arm64)
elseif(IOS_ARCH STREQUAL "x86_64")
    set(CMAKE_OSX_ARCHITECTURES x86_64)
endif()
```

### 3. 构建目标管理

#### iOS 特定 CMake 配置
```cmake
# 主 CMakeLists.txt 中的 iOS 检测
if(IOS)
    set(DIST_DIR ios)
    set(INSTALL_TYPE ARCHIVE)

    # iOS 特定编译选项
    set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -fembed-bitcode")
    set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -fembed-bitcode")

    # 禁用不需要的功能
    set(FILAMENT_SKIP_SAMPLES ON)
    set(FILAMENT_SUPPORTS_OSMESA OFF)
endif()
```

## 构建流程详解

### 1. 桌面工具构建

**目的**: 构建 matc、resgen 等工具，用于资源编译

```bash
# 构建脚本中的桌面构建函数
function build_desktop {
    echo "Building desktop tools..."

    mkdir -p out/cmake-release
    cd out/cmake-release

    cmake \
        -G Ninja \
        -DCMAKE_BUILD_TYPE=Release \
        -DCMAKE_INSTALL_PREFIX=../release/filament \
        -DFILAMENT_ENABLE_LTO=OFF \
        ../..

    ninja
    ninja install
}
```

### 2. iOS 库编译

#### 设备版本编译
```bash
function build_ios_target {
    local target=$1      # Debug/Release
    local arch=$2        # arm64
    local platform=$3    # iphoneos

    echo "Building iOS ${target} (${arch}) for ${platform}..."

    mkdir -p "out/cmake-ios-${target}-${arch}"
    cd "out/cmake-ios-${target}-${arch}"

    cmake \
        -G Ninja \
        -DIMPORT_EXECUTABLES_DIR=out \
        -DCMAKE_BUILD_TYPE=${target} \
        -DCMAKE_INSTALL_PREFIX="../ios-${target}/filament" \
        -DIOS_ARCH=${arch} \
        -DPLATFORM_NAME=${platform} \
        -DIOS=1 \
        -DCMAKE_TOOLCHAIN_FILE=../../third_party/clang/iOS.cmake \
        ../..

    ninja
    ninja install
}
```

#### 模拟器版本编译
```bash
# 模拟器特定配置
if [[ "${ISSUE_SIMULATOR_BUILD}" == "true" ]]; then
    build_ios_target ${target} "x86_64" "iphonesimulator"
fi
```

### 3. 通用库创建 (Universal Binary)

**功能**: 使用 lipo 工具合并不同架构的库文件

```bash
function create_universal_libraries {
    local target_dir="out/ios-${config}/filament/lib"

    mkdir -p "${target_dir}/universal"

    # 合并 arm64 和 x86_64 架构
    for lib in libfilament.a libbackend.a libutils.a; do
        lipo -create \
            "${target_dir}/arm64/${lib}" \
            "${target_dir}/x86_64/${lib}" \
            -output "${target_dir}/universal/${lib}"
    done

    echo "Universal libraries created in ${target_dir}/universal/"
}
```

### 2. 预编译二进制

#### 下载预编译库
```bash
# 自动下载预编译的 iOS 库
function download_prebuilt_ios {
    local version="1.65.0"
    local url="https://github.com/google/filament/releases/download/v${version}/filament-v${version}-ios.tgz"

    curl -L "${url}" -o filament-ios.tgz
    tar -xzf filament-ios.tgz
    mv filament-v${version}-ios out/filament-ios
}
```

## Xcode 项目生成

### 1. XcodeGen 配置

**位置**: `ios/samples/app-template.yml`
**功能**: 统一的 Xcode 项目模板

```yaml
# app-template.yml
name: ${sample_name}

options:
  bundleIdPrefix: com.google.filament
  deploymentTarget:
    iOS: 11.0
  groupSortPosition: top

settings:
  base:
    CLANG_CXX_LANGUAGE_STANDARD: c++14
    CLANG_CXX_LIBRARY: libc++
    ENABLE_BITCODE: NO
    IPHONEOS_DEPLOYMENT_TARGET: 11.0
    SKIP_INSTALL: YES

configurations:
  Debug:
    LIBRARY_SEARCH_PATHS:
      - $(PROJECT_DIR)/../../out/ios-debug/filament/lib/arm64
      - $(PROJECT_DIR)/../../out/ios-debug/filament/lib/x86_64
  Release:
    LIBRARY_SEARCH_PATHS:
      - $(PROJECT_DIR)/../../out/ios-release/filament/lib/arm64
      - $(PROJECT_DIR)/../../out/ios-release/filament/lib/x86_64

schemes:
  ${sample_name} Metal:
    build:
      targets:
        ${sample_name}: all
    run:
      config: Debug
      environmentVariables:
        FILAMENT_BACKEND: metal

  ${sample_name} OpenGL:
    build:
      targets:
        ${sample_name}: all
    run:
      config: Debug
      environmentVariables:
        FILAMENT_BACKEND: opengl
```

### 2. 项目自动生成

```bash
#!/bin/bash
# generate-samples.sh

# 为每个示例生成 Xcode 项目
for sample in hello-triangle hello-pbr hello-gltf gltf-viewer; do
    echo "Generating ${sample} Xcode project..."

    cd "ios/samples/${sample}"

    # 替换模板变量
    sed "s/\${sample_name}/${sample}/g" ../app-template.yml > project.yml

    # 生成 Xcode 项目
    xcodegen generate

    echo "Generated ${sample}.xcodeproj"

    cd - > /dev/null
done
```

## 构建问题排查

### 1. 常见构建错误

#### SDK 版本问题
```bash
# 检查可用的 iOS SDK
xcrun --show-sdk-version --sdk iphoneos

# 设置特定 SDK 版本
export IPHONEOS_DEPLOYMENT_TARGET=11.0
```

#### 工具链问题
```bash
# 检查 Xcode 命令行工具
xcode-select --print-path

# 重置到最新 Xcode
sudo xcode-select --reset
```

### 2. 清理构建缓存

```bash
# 完全清理构建
rm -rf out/
./build.sh -c -p ios debug

# 仅清理 CMake 缓存
find out/ -name "CMakeCache.txt" -delete
find out/ -name "CMakeFiles" -type d -exec rm -rf {} +
```

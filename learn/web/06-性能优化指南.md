# Filament Web 性能优化指南

## 1. 性能优化策略概览

Filament Web 的性能优化需要从多个层面入手，包括引擎配置、资源管理、渲染优化、内存管理和网络优化。本指南提供系统性的优化方案。

### 1.1 性能优化金字塔

```
┌─────────────────────────────────────┐
│           应用层优化                 │  ← 算法和逻辑优化
├─────────────────────────────────────┤
│           引擎层优化                 │  ← 渲染设置和特性配置
├─────────────────────────────────────┤
│          资源层优化                  │  ← 纹理、模型、材质优化
├─────────────────────────────────────┤
│          系统层优化                  │  ← 内存、网络、缓存优化
└─────────────────────────────────────┘
```

```javascript
// 渲染管道优化配置
class RenderPipelineOptimizer {
    constructor(engine, view, scene) {
        this.engine = engine;
        this.view = view;
        this.scene = scene;
        this.renderables = [];
        this.frameStats = {
            drawCalls: 0,
            triangles: 0,
            vertices: 0
        };
    }

    optimizeRenderSettings() {
        // 1. 启用基础剔除
        this.enableCullingOptimizations();

        // 2. 配置阴影优化
        this.optimizeShadowSettings();

        // 3. 设置后处理优化
        this.optimizePostProcessing();

        // 4. 配置渲染质量
        this.configureQualitySettings();
    }

    enableCullingOptimizations() {
        // 视锥体剔除 (自动启用)
        // 背面剔除
        const renderables = this.scene.getRenderableManager();

        this.renderables.forEach(entity => {
            const instance = renderables.getInstance(entity);
            if (instance) {
                // 启用背面剔除
                renderables.setCulling(instance, true);

                // 设置包围盒用于剔除
                const boundingBox = this.calculateBoundingBox(entity);
                renderables.setAxisAlignedBoundingBox(instance, boundingBox);
            }
        });
    }

    optimizeShadowSettings() {
        const shadowOptions = this.view.getShadowOptions();

        // 根据设备能力调整阴影分辨率
        const deviceCapability = this.assessDeviceCapability();

        if (deviceCapability === 'high') {
            shadowOptions.mapSize = 2048;
            shadowOptions.shadowCascades = 4;
        } else if (deviceCapability === 'medium') {
            shadowOptions.mapSize = 1024;
            shadowOptions.shadowCascades = 3;
        } else {
            shadowOptions.mapSize = 512;
            shadowOptions.shadowCascades = 2;
        }

        // 优化阴影偏移
        shadowOptions.constantBias = 0.001;
        shadowOptions.normalBias = 1.0;

        // 启用阴影采样优化
        shadowOptions.screenSpaceContactShadows = deviceCapability === 'high';

        this.view.setShadowOptions(shadowOptions);
    }

    optimizePostProcessing() {
        const options = this.view.getPostProcessOptions();

        // 基于设备能力调整后处理
        const capability = this.assessDeviceCapability();

        if (capability === 'high') {
            // 高端设备：启用全部特效
            options.antiAliasing = Filament.View$AntiAliasing.FXAA;
            options.bloomOptions.enabled = true;
            options.bloomOptions.strength = 0.1;
            options.vignetteOptions.enabled = true;

            // 色调映射
            options.colorGrading.enabled = true;
            options.colorGrading.toneMapping = Filament.View$ToneMapping.ACES;
        } else if (capability === 'medium') {
            // 中端设备：基础特效
            options.antiAliasing = Filament.View$AntiAliasing.FXAA;
            options.bloomOptions.enabled = false;
            options.vignetteOptions.enabled = false;

            options.colorGrading.enabled = true;
            options.colorGrading.toneMapping = Filament.View$ToneMapping.LINEAR;
        } else {
            // 低端设备：禁用后处理
            options.antiAliasing = Filament.View$AntiAliasing.NONE;
            options.bloomOptions.enabled = false;
            options.vignetteOptions.enabled = false;
            options.colorGrading.enabled = false;
        }

        this.view.setPostProcessOptions(options);
    }

    // 自适应质量控制
    configureQualitySettings() {
        const qualityManager = new AdaptiveQualityManager(this.engine);
        qualityManager.setTargetFrameRate(60);
        qualityManager.setQualityBounds('low', 'high');

        // 注册质量调整回调
        qualityManager.onQualityChange((newQuality) => {
            this.adjustRenderingQuality(newQuality);
        });

        return qualityManager;
    }

    adjustRenderingQuality(quality) {
        switch (quality) {
            case 'high':
                this.setRenderScale(1.0);
                this.setShadowQuality('high');
                this.setTextureQuality('high');
                break;

            case 'medium':
                this.setRenderScale(0.8);
                this.setShadowQuality('medium');
                this.setTextureQuality('medium');
                break;

            case 'low':
                this.setRenderScale(0.6);
                this.setShadowQuality('low');
                this.setTextureQuality('low');
                break;
        }
    }

    // 设备能力评估
    assessDeviceCapability() {
        const gl = this.engine.getGL();
        const renderer = gl.getParameter(gl.RENDERER);
        const vendor = gl.getParameter(gl.VENDOR);

        // 基于 GPU 信息评估设备能力
        const gpuTier = this.classifyGPU(renderer, vendor);

        // 基于内存评估
        const memoryTier = this.classifyMemory();

        // 综合评估
        return this.combineAssessments(gpuTier, memoryTier);
    }

    classifyGPU(renderer, vendor) {
        const highEndGPUs = [
            'RTX', 'GTX 1060', 'GTX 1070', 'GTX 1080',
            'RX 580', 'RX 590', 'Vega',
            'Apple M1', 'Apple M2'
        ];

        const midRangeGPUs = [
            'GTX 960', 'GTX 1050',
            'RX 560', 'RX 570',
            'Intel Iris'
        ];

        const gpuString = renderer.toUpperCase();

        if (highEndGPUs.some(gpu => gpuString.includes(gpu.toUpperCase()))) {
            return 'high';
        } else if (midRangeGPUs.some(gpu => gpuString.includes(gpu.toUpperCase()))) {
            return 'medium';
        } else {
            return 'low';
        }
    }

    classifyMemory() {
        if (performance.memory) {
            const totalMemory = performance.memory.jsHeapSizeLimit;

            if (totalMemory > 4 * 1024 * 1024 * 1024) { // 4GB+
                return 'high';
            } else if (totalMemory > 2 * 1024 * 1024 * 1024) { // 2GB+
                return 'medium';
            } else {
                return 'low';
            }
        }

        return 'medium'; // 默认值
    }
}

// 自适应质量管理器
class AdaptiveQualityManager {
    constructor(engine) {
        this.engine = engine;
        this.targetFrameRate = 60;
        this.currentQuality = 'medium';
        this.frameTimeHistory = [];
        this.adjustmentCooldown = 0;
        this.callbacks = [];
    }

    setTargetFrameRate(fps) {
        this.targetFrameRate = fps;
        this.targetFrameTime = 1000 / fps; // 目标帧时间 (ms)
    }

    update(deltaTime) {
        this.frameTimeHistory.push(deltaTime);

        // 保持最近 60 帧的历史
        if (this.frameTimeHistory.length > 60) {
            this.frameTimeHistory.shift();
        }

        // 冷却期间不进行调整
        if (this.adjustmentCooldown > 0) {
            this.adjustmentCooldown--;
            return;
        }

        // 计算平均帧时间
        const avgFrameTime = this.frameTimeHistory.reduce((a, b) => a + b, 0)
                           / this.frameTimeHistory.length;

        // 判断是否需要调整质量
        if (avgFrameTime > this.targetFrameTime * 1.2) {
            // 性能不足，降低质量
            this.decreaseQuality();
        } else if (avgFrameTime < this.targetFrameTime * 0.8) {
            // 性能充足，提高质量
            this.increaseQuality();
        }
    }

    decreaseQuality() {
        let newQuality = this.currentQuality;

        if (this.currentQuality === 'high') {
            newQuality = 'medium';
        } else if (this.currentQuality === 'medium') {
            newQuality = 'low';
        }

        if (newQuality !== this.currentQuality) {
            this.setQuality(newQuality);
            this.adjustmentCooldown = 120; // 2秒冷却
        }
    }

    increaseQuality() {
        let newQuality = this.currentQuality;

        if (this.currentQuality === 'low') {
            newQuality = 'medium';
        } else if (this.currentQuality === 'medium') {
            newQuality = 'high';
        }

        if (newQuality !== this.currentQuality) {
            this.setQuality(newQuality);
            this.adjustmentCooldown = 180; // 3秒冷却
        }
    }

    setQuality(quality) {
        this.currentQuality = quality;
        this.callbacks.forEach(callback => callback(quality));
        console.log(`质量调整为: ${quality}`);
    }

    onQualityChange(callback) {
        this.callbacks.push(callback);
    }
}
```

```javascript
// 几何优化管理器
class GeometryOptimizer {
    constructor(engine) {
        this.engine = engine;
        this.lodLevels = new Map();
        this.instancedObjects = new Map();
        this.frustumCuller = new FrustumCuller();
    }

    // 细节层次 (LOD) 优化
    setupLevelOfDetail(entity, lodConfigs) {
        const lodSystem = {
            entity: entity,
            levels: lodConfigs.map(config => ({
                distance: config.distance,
                geometry: config.geometry,
                active: false
            })),
            currentLevel: -1
        };

        this.lodLevels.set(entity, lodSystem);
    }

    updateLevelOfDetail(cameraPosition) {
        for (const [entity, lodSystem] of this.lodLevels.entries()) {
            const entityPosition = this.getEntityPosition(entity);
            const distance = this.calculateDistance(cameraPosition, entityPosition);

            // 选择合适的 LOD 级别
            let selectedLevel = -1;
            for (let i = 0; i < lodSystem.levels.length; i++) {
                if (distance <= lodSystem.levels[i].distance) {
                    selectedLevel = i;
                    break;
                }
            }

            // 更新几何体
            if (selectedLevel !== lodSystem.currentLevel) {
                this.switchLODLevel(entity, lodSystem, selectedLevel);
                lodSystem.currentLevel = selectedLevel;
            }
        }
    }

    switchLODLevel(entity, lodSystem, newLevel) {
        const renderables = this.engine.getRenderableManager();
        const instance = renderables.getInstance(entity);

        if (instance) {
            if (newLevel >= 0 && newLevel < lodSystem.levels.length) {
                const level = lodSystem.levels[newLevel];
                // 更新几何体
                renderables.setGeometryAt(instance, 0,
                    level.geometry.primitiveType,
                    level.geometry.vertexBuffer,
                    level.geometry.indexBuffer,
                    level.geometry.offset,
                    level.geometry.count);
            } else {
                // 距离过远，隐藏对象
                renderables.setLayerMask(instance, 0, 0x00);
            }
        }
    }

    // 实例化渲染优化
    setupInstancedRendering(baseGeometry, transforms) {
        // 创建实例化缓冲区
        const instanceBuffer = this.createInstanceBuffer(transforms);

        // 创建实例化材质
        const instancedMaterial = this.createInstancedMaterial();

        // 配置实例化渲染
        const entity = Filament.EntityManager.get().create();

        Filament.RenderableManager$Builder(1)
            .boundingBox(this.calculateInstancedBoundingBox(transforms))
            .material(0, instancedMaterial)
            .geometry(0, baseGeometry.primitiveType,
                baseGeometry.vertexBuffer, baseGeometry.indexBuffer)
            .instances(transforms.length)
            .build(this.engine, entity);

        return entity;
    }

    createInstanceBuffer(transforms) {
        const instanceCount = transforms.length;
        const instanceData = new Float32Array(instanceCount * 16); // 4x4 矩阵

        transforms.forEach((transform, index) => {
            const offset = index * 16;
            instanceData.set(transform, offset);
        });

        // 创建实例化顶点缓冲区
        const instanceBuffer = Filament.VertexBuffer$Builder()
            .vertexCount(instanceCount)
            .bufferCount(1)
            .attribute(Filament.VertexAttribute.BONE_MATRICES, 0,
                Filament.VertexBuffer$AttributeType.MAT4, 0, 64)
            .normalized(false)
            .build(this.engine);

        const bufferData = new Uint8Array(instanceData.buffer);
        instanceBuffer.setBufferAt(this.engine, 0, bufferData);

        return instanceBuffer;
    }

    // 批处理优化
    batchSimilarObjects(renderableEntities) {
        const batches = new Map();

        // 按材质和几何体分组
        renderableEntities.forEach(entity => {
            const renderable = this.getRenderableData(entity);
            const key = `${renderable.materialId}_${renderable.geometryId}`;

            if (!batches.has(key)) {
                batches.set(key, []);
            }
            batches.get(key).push(entity);
        });

        // 为每个批次创建实例化渲染
        const batchedEntities = [];
        for (const [key, entities] of batches.entries()) {
            if (entities.length > 1) {
                const transforms = entities.map(e => this.getEntityTransform(e));
                const batchEntity = this.setupInstancedRendering(
                    this.getEntityGeometry(entities[0]),
                    transforms
                );
                batchedEntities.push(batchEntity);

                // 隐藏原始实体
                entities.forEach(e => this.hideEntity(e));
            }
        }

        return batchedEntities;
    }

    // 视锥体剔除
    performFrustumCulling(camera, renderables) {
        const frustum = this.frustumCuller.extractFrustum(camera);
        const visibleRenderables = [];

        renderables.forEach(entity => {
            const boundingBox = this.getEntityBoundingBox(entity);
            if (this.frustumCuller.testAABB(frustum, boundingBox)) {
                visibleRenderables.push(entity);
            }
        });

        return visibleRenderables;
    }

    // 工具方法
    calculateDistance(pos1, pos2) {
        const dx = pos1[0] - pos2[0];
        const dy = pos1[1] - pos2[1];
        const dz = pos1[2] - pos2[2];
        return Math.sqrt(dx * dx + dy * dy + dz * dz);
    }

    getEntityPosition(entity) {
        const tm = this.engine.getTransformManager();
        const instance = tm.getInstance(entity);
        const transform = tm.getTransform(instance);
        return [transform[12], transform[13], transform[14]];
    }
}

// 视锥体剔除器
class FrustumCuller {
    extractFrustum(camera) {
        // 提取相机视锥体平面
        const viewMatrix = camera.getViewMatrix();
        const projMatrix = camera.getProjectionMatrix();

        // 计算视锥体的6个平面
        const frustum = this.calculateFrustumPlanes(viewMatrix, projMatrix);

        return frustum;
    }

    calculateFrustumPlanes(view, proj) {
        // 计算 MVP 矩阵
        const mvp = this.multiplyMatrices(proj, view);

        // 提取6个平面: left, right, bottom, top, near, far
        const planes = [];

        // Left plane
        planes.push(this.normalizePlane([
            mvp[3] + mvp[0],
            mvp[7] + mvp[4],
            mvp[11] + mvp[8],
            mvp[15] + mvp[12]
        ]));

        // Right plane
        planes.push(this.normalizePlane([
            mvp[3] - mvp[0],
            mvp[7] - mvp[4],
            mvp[11] - mvp[8],
            mvp[15] - mvp[12]
        ]));

        // Bottom plane
        planes.push(this.normalizePlane([
            mvp[3] + mvp[1],
            mvp[7] + mvp[5],
            mvp[11] + mvp[9],
            mvp[15] + mvp[13]
        ]));

        // Top plane
        planes.push(this.normalizePlane([
            mvp[3] - mvp[1],
            mvp[7] - mvp[5],
            mvp[11] - mvp[9],
            mvp[15] - mvp[13]
        ]));

        // Near plane
        planes.push(this.normalizePlane([
            mvp[3] + mvp[2],
            mvp[7] + mvp[6],
            mvp[11] + mvp[10],
            mvp[15] + mvp[14]
        ]));

        // Far plane
        planes.push(this.normalizePlane([
            mvp[3] - mvp[2],
            mvp[7] - mvp[6],
            mvp[11] - mvp[10],
            mvp[15] - mvp[14]
        ]));

        return planes;
    }

    testAABB(frustumPlanes, boundingBox) {
        const { center, halfExtent } = boundingBox;

        for (const plane of frustumPlanes) {
            const [a, b, c, d] = plane;

            // 计算 AABB 与平面的距离
            const distance = a * center[0] + b * center[1] + c * center[2] + d;
            const radius = Math.abs(a * halfExtent[0]) +
                          Math.abs(b * halfExtent[1]) +
                          Math.abs(c * halfExtent[2]);

            if (distance + radius < 0) {
                return false; // 完全在平面后面
            }
        }

        return true; // 与视锥体相交或在内部
    }

    normalizePlane(plane) {
        const [a, b, c, d] = plane;
        const length = Math.sqrt(a * a + b * b + c * c);
        return [a / length, b / length, c / length, d / length];
    }

    multiplyMatrices(a, b) {
        const result = new Float32Array(16);

        for (let i = 0; i < 4; i++) {
            for (let j = 0; j < 4; j++) {
                result[i * 4 + j] =
                    a[i * 4 + 0] * b[0 * 4 + j] +
                    a[i * 4 + 1] * b[1 * 4 + j] +
                    a[i * 4 + 2] * b[2 * 4 + j] +
                    a[i * 4 + 3] * b[3 * 4 + j];
            }
        }

        return result;
    }
}
```

```javascript
// 纹理优化管理器
class TextureOptimizer {
    constructor(engine) {
        this.engine = engine;
        this.textureCache = new Map();
        this.compressionSupport = this.detectCompressionSupport();
        this.qualitySettings = this.getQualitySettings();
    }

    detectCompressionSupport() {
        const gl = this.engine.getGL();

        return {
            s3tc: !!gl.getExtension('WEBGL_compressed_texture_s3tc'),
            etc: !!gl.getExtension('WEBGL_compressed_texture_etc'),
            astc: !!gl.getExtension('WEBGL_compressed_texture_astc'),
            pvrtc: !!gl.getExtension('WEBGL_compressed_texture_pvrtc'),
            bptc: !!gl.getExtension('EXT_texture_compression_bptc')
        };
    }

    getQualitySettings() {
        const deviceCapability = this.assessDeviceCapability();

        const settings = {
            high: {
                maxTextureSize: 2048,
                anisotropicFiltering: 16,
                mipmapLevels: -1, // 全部级别
                compressionQuality: 'high'
            },
            medium: {
                maxTextureSize: 1024,
                anisotropicFiltering: 4,
                mipmapLevels: 8,
                compressionQuality: 'medium'
            },
            low: {
                maxTextureSize: 512,
                anisotropicFiltering: 1,
                mipmapLevels: 4,
                compressionQuality: 'low'
            }
        };

        return settings[deviceCapability];
    }

    // 智能纹理格式选择
    selectOptimalFormat(textureData) {
        const { width, height, hasAlpha, isNormalMap } = textureData;

        // 基于内容类型选择格式
        if (isNormalMap) {
            return this.selectNormalMapFormat();
        }

        if (hasAlpha) {
            return this.selectAlphaFormat();
        }

        return this.selectOpaqueFormat();
    }

    selectNormalMapFormat() {
        if (this.compressionSupport.bptc) {
            return {
                format: Filament.Texture$InternalFormat.COMPRESSED_RGBA_BPTC_UNORM,
                type: 'bptc'
            };
        }

        if (this.compressionSupport.s3tc) {
            return {
                format: Filament.Texture$InternalFormat.COMPRESSED_RGBA_S3TC_DXT5,
                type: 's3tc'
            };
        }

        return {
            format: Filament.Texture$InternalFormat.RG8,
            type: 'uncompressed'
        };
    }

    selectAlphaFormat() {
        if (this.compressionSupport.s3tc) {
            return {
                format: Filament.Texture$InternalFormat.COMPRESSED_RGBA_S3TC_DXT5,
                type: 's3tc'
            };
        }

        if (this.compressionSupport.etc) {
            return {
                format: Filament.Texture$InternalFormat.COMPRESSED_RGBA8_ETC2_EAC,
                type: 'etc2'
            };
        }

        return {
            format: Filament.Texture$InternalFormat.RGBA8,
            type: 'uncompressed'
        };
    }

    selectOpaqueFormat() {
        if (this.compressionSupport.s3tc) {
            return {
                format: Filament.Texture$InternalFormat.COMPRESSED_RGB_S3TC_DXT1,
                type: 's3tc'
            };
        }

        if (this.compressionSupport.etc) {
            return {
                format: Filament.Texture$InternalFormat.COMPRESSED_RGB8_ETC2,
                type: 'etc2'
            };
        }

        return {
            format: Filament.Texture$InternalFormat.RGB8,
            type: 'uncompressed'
        };
    }

    // 动态纹理缩放
    scaleTextureForDevice(imageData) {
        const maxSize = this.qualitySettings.maxTextureSize;
        let { width, height } = imageData;

        // 如果纹理过大，进行缩放
        if (width > maxSize || height > maxSize) {
            const scale = Math.min(maxSize / width, maxSize / height);
            width = Math.floor(width * scale);
            height = Math.floor(height * scale);

            return this.resizeImage(imageData, width, height);
        }

        return imageData;
    }

    resizeImage(imageData, newWidth, newHeight) {
        const canvas = document.createElement('canvas');
        const ctx = canvas.getContext('2d');

        canvas.width = newWidth;
        canvas.height = newHeight;

        // 使用高质量缩放
        ctx.imageSmoothingEnabled = true;
        ctx.imageSmoothingQuality = 'high';

        ctx.drawImage(imageData, 0, 0, newWidth, newHeight);

        return canvas;
    }

    // 纹理流送系统
    async createStreamingTexture(baseUrl, levels) {
        const streamingTexture = {
            baseUrl: baseUrl,
            levels: levels,
            currentLevel: -1,
            texture: null,
            loading: false
        };

        // 首先加载最低质量版本
        await this.loadTextureLevel(streamingTexture, 0);

        // 异步加载更高质量版本
        this.progressivelyLoadHigherLevels(streamingTexture);

        return streamingTexture;
    }

    async loadTextureLevel(streamingTexture, level) {
        if (level >= streamingTexture.levels.length) return;

        streamingTexture.loading = true;
        const levelData = streamingTexture.levels[level];

        try {
            const imageData = await this.loadImageFromURL(levelData.url);
            const processedImage = this.scaleTextureForDevice(imageData);

            if (streamingTexture.texture) {
                // 更新现有纹理
                this.updateTexture(streamingTexture.texture, processedImage);
            } else {
                // 创建新纹理
                streamingTexture.texture = this.createTexture(processedImage);
            }

            streamingTexture.currentLevel = level;
            console.log(`纹理级别 ${level} 加载完成`);

        } catch (error) {
            console.error(`纹理级别 ${level} 加载失败:`, error);
        } finally {
            streamingTexture.loading = false;
        }
    }

    progressivelyLoadHigherLevels(streamingTexture) {
        let nextLevel = streamingTexture.currentLevel + 1;

        const loadNext = async () => {
            if (nextLevel < streamingTexture.levels.length && !streamingTexture.loading) {
                await this.loadTextureLevel(streamingTexture, nextLevel);
                nextLevel++;

                // 继续加载下一级别
                setTimeout(loadNext, 100);
            }
        };

        setTimeout(loadNext, 500); // 延迟开始高质量加载
    }

    // 纹理内存管理
    manageTextureMemory() {
        const memoryUsage = this.calculateTextureMemoryUsage();
        const memoryLimit = this.getMemoryLimit();

        if (memoryUsage > memoryLimit * 0.8) {
            console.log('纹理内存使用率过高，开始清理...');
            this.performMemoryCleanup();
        }
    }

    calculateTextureMemoryUsage() {
        let totalMemory = 0;

        for (const texture of this.textureCache.values()) {
            totalMemory += this.calculateTextureSize(texture);
        }

        return totalMemory;
    }

    calculateTextureSize(texture) {
        // 简化的纹理大小计算
        const { width, height, format, mipLevels } = texture;
        let bytesPerPixel = 4; // 默认 RGBA

        switch (format) {
            case 'rgb8':
                bytesPerPixel = 3;
                break;
            case 'rg8':
                bytesPerPixel = 2;
                break;
            case 'r8':
                bytesPerPixel = 1;
                break;
            case 'compressed':
                bytesPerPixel = 1; // 压缩纹理平均值
                break;
        }

        let totalSize = width * height * bytesPerPixel;

        // 包含 mipmap 的大小
        if (mipLevels > 1) {
            totalSize *= 1.33; // mipmap 额外占用约 33% 内存
        }

        return totalSize;
    }

    performMemoryCleanup() {
        // 按最后使用时间排序
        const texturesByUsage = Array.from(this.textureCache.entries())
            .sort((a, b) => a[1].lastUsed - b[1].lastUsed);

        // 清理最老的 25% 纹理
        const cleanupCount = Math.floor(texturesByUsage.length * 0.25);

        for (let i = 0; i < cleanupCount; i++) {
            const [key, texture] = texturesByUsage[i];
            texture.texture.delete();
            this.textureCache.delete(key);
        }

        console.log(`清理了 ${cleanupCount} 个纹理`);
    }
}
```

```javascript
class GeometryDataOptimizer {
    constructor() {

        const compressionRatio = originalSize / compressedSize;

        console.log(`几何数据压缩: ${originalSize} -> ${compressedSize} bytes (${compressionRatio.toFixed(2)}x)`);

        return compressed;

    quantizePositions(positions) {
        // 计算包围盒
        const bounds = this.calculateBounds(positions);
        const scale = this.calculateQuantizationScale(bounds, this.quantizationLevels.position);

        const quantized = new Int16Array(positions.length);

        for (let i = 0; i < positions.length; i += 3) {
            quantized[i] = Math.round((positions[i] - bounds.min[0]) * scale[0]);
            quantized[i + 1] = Math.round((positions[i + 1] - bounds.min[1]) * scale[1]);
            quantized[i + 2] = Math.round((positions[i + 2] - bounds.min[2]) * scale[2]);
        }

        return {
            data: quantized,
            bounds: bounds,
            scale: scale,
            type: 'quantized_positions'
        };
    }

    compressNormals(normals) {
        // 使用 Octahedral 编码压缩法向量
        const compressed = new Int16Array(normals.length / 3 * 2);

        for (let i = 0, j = 0; i < normals.length; i += 3, j += 2) {
            const [x, y, z] = [normals[i], normals[i + 1], normals[i + 2]];
            const octahedral = this.encodeOctahedral(x, y, z);

            compressed[j] = Math.round(octahedral[0] * 32767);
            compressed[j + 1] = Math.round(octahedral[1] * 32767);
        }

        return {
            data: compressed,
            type: 'octahedral_normals'
        };
    }

    encodeOctahedral(x, y, z) {
        // 将单位向量编码为八面体坐标
        const l1norm = Math.abs(x) + Math.abs(y) + Math.abs(z);
        const octX = x / l1norm;
        const octY = y / l1norm;

        if (z >= 0) {
            return [octX, octY];
        } else {
            return [
                (1 - Math.abs(octY)) * Math.sign(octX),
                (1 - Math.abs(octX)) * Math.sign(octY)
            ];
        }
    }

    // 几何 LOD 生成
    generateLODLevels(geometry, levels = [0.75, 0.5, 0.25]) {
        const lodGeometries = [geometry]; // 原始几何体作为最高 LOD

        levels.forEach(factor => {
            const simplifiedGeometry = this.simplifyGeometry(geometry, factor);
            lodGeometries.push(simplifiedGeometry);
        });

        return lodGeometries;
    }

    simplifyGeometry(geometry, factor) {
        // 边折叠简化算法
        const quadricErrors = this.calculateQuadricErrors(geometry);
        const edges = this.buildEdgeList(geometry);

        // 按错误度量排序边
        edges.sort((a, b) => a.error - b.error);

        const targetVertexCount = Math.floor(geometry.vertexCount * factor);
        const simplifiedGeometry = this.collapseEdges(geometry, edges, targetVertexCount);

        return simplifiedGeometry;

    calculateQuadricErrors(geometry) {

        for (let i = 0; i < geometry.vertexCount; i++) {

                geometry.indices[i + 1],
                geometry.indices[i + 2]

            const quadric = this.planeToQuadric(plane);

            // 将平面误差添加到三个顶点
            this.addQuadric(quadrics[v0], quadric);
            this.addQuadric(quadrics[v1], quadric);
            this.addQuadric(quadrics[v2], quadric);
        }

        return quadrics;
    }

        const deduplicatedGeometry = this.removeDuplicateVertices(geometry);

        // 2. 优化顶点缓存

    removeDuplicateVertices(geometry) {

        const indexMapping = [];

        for (let i = 0; i < geometry.vertexCount; i++) {
            const vertex = this.getVertexData(geometry, i);
            const hash = this.hashVertex(vertex);

            if (vertexMap.has(hash)) {
                indexMapping[i] = vertexMap.get(hash);
            } else {
                const newIndex = uniqueVertices.length;
                uniqueVertices.push(vertex);
                vertexMap.set(hash, newIndex);
                indexMapping[i] = newIndex;
            }
        }

        // 重新映射索引
        const newIndices = geometry.indices.map(index => indexMapping[index]);

        console.log(`顶点去重: ${geometry.vertexCount} -> ${uniqueVertices.length}`);

        return {
            positions: this.extractVertexComponent(uniqueVertices, 'position'),
            normals: this.extractVertexComponent(uniqueVertices, 'normal'),
            uvs: this.extractVertexComponent(uniqueVertices, 'uv'),

            vertexCount: uniqueVertices.length

    optimizeVertexCache(geometry) {
        // 使用 Tom Forsyth 的顶点缓存优化算法
        const optimizedIndices = this.forsythOptimize(geometry.indices);

        return {
            ...geometry,
            indices: optimizedIndices
        };
    }

    forsythOptimize(indices) {
        const triangleCount = indices.length / 3;
        const vertexCount = Math.max(...indices) + 1;

        // 初始化顶点数据
        const vertices = new Array(vertexCount).fill(0).map(() => ({
            cachePosition: -1,
            score: 0,
            triangleCount: 0,
            triangles: []
        }));

        // 构建顶点-三角形邻接关系
        for (let i = 0; i < triangleCount; i++) {
            const triangle = [indices[i * 3], indices[i * 3 + 1], indices[i * 3 + 2]];

            triangle.forEach(vertexIndex => {
                vertices[vertexIndex].triangles.push(i);
                vertices[vertexIndex].triangleCount++;
            });
        }

        // 执行优化
        const optimizedIndices = new Array(indices.length);
        const triangleEmitted = new Array(triangleCount).fill(false);
        let outputIndex = 0;

        while (outputIndex < indices.length) {
            // 选择得分最高的三角形
            let bestTriangle = -1;
            let bestScore = -1;

            for (let i = 0; i < triangleCount; i++) {
                if (!triangleEmitted[i]) {
                    const score = this.calculateTriangleScore(i, indices, vertices);
                    if (score > bestScore) {
                        bestScore = score;
                        bestTriangle = i;
                    }
                }
            }

            // 输出三角形
            if (bestTriangle >= 0) {
                triangleEmitted[bestTriangle] = true;
                const triangle = [
                    indices[bestTriangle * 3],
                    indices[bestTriangle * 3 + 1],
                    indices[bestTriangle * 3 + 2]
                ];

                triangle.forEach(vertexIndex => {
                    optimizedIndices[outputIndex++] = vertexIndex;
                });

                // 更新顶点缓存
                this.updateVertexCache(triangle, vertices);
            } else {
                break; // 无法找到更多三角形
            }
        }

        return optimizedIndices;
    }

    // 工具方法
    calculateBounds(positions) {
        const min = [Infinity, Infinity, Infinity];
        const max = [-Infinity, -Infinity, -Infinity];

        for (let i = 0; i < positions.length; i += 3) {
            min[0] = Math.min(min[0], positions[i]);
            min[1] = Math.min(min[1], positions[i + 1]);
            min[2] = Math.min(min[2], positions[i + 2]);

            max[0] = Math.max(max[0], positions[i]);
            max[1] = Math.max(max[1], positions[i + 1]);
            max[2] = Math.max(max[2], positions[i + 2]);
        }

        return { min, max };
    }

    calculateQuantizationScale(bounds, bits) {
        const range = [
            bounds.max[0] - bounds.min[0],
            bounds.max[1] - bounds.min[1],
            bounds.max[2] - bounds.min[2]
        ];

        const maxValue = (1 << bits) - 1;

        return [
            range[0] > 0 ? maxValue / range[0] : 1,
            range[1] > 0 ? maxValue / range[1] : 1,
            range[2] > 0 ? maxValue / range[2] : 1
        ];
    }

    hashVertex(vertex) {
        // 简单的顶点哈希函数
        return `${vertex.position.join(',')}_${vertex.normal.join(',')}_${vertex.uv.join(',')}`;
    }
```

```javascript
class MemoryPoolManager {
    constructor(engine) {
        this.engine = engine;
        this.pools = new Map();
        this.memoryBudget = this.calculateMemoryBudget();
        this.currentUsage = 0;
        this.gcThreshold = 0.8; // 80% 使用率触发 GC

        this.setupMemoryMonitoring();
    }

    calculateMemoryBudget() {
        if (performance.memory) {
            // 使用可用内存的 60%
            return performance.memory.jsHeapSizeLimit * 0.6;
        }

        // 默认 1GB 预算
        return 1024 * 1024 * 1024;
    }

    // 创建对象池
    createPool(type, factory, options = {}) {
        const pool = {
            type: type,
            factory: factory,
            available: [],
            inUse: new Set(),
            maxSize: options.maxSize || 100,
            prealloc: options.prealloc || 10,
            resetFunction: options.resetFunction || null
        };

        // 预分配对象
        for (let i = 0; i < pool.prealloc; i++) {
            const obj = pool.factory();
            pool.available.push(obj);
        }

        this.pools.set(type, pool);
        console.log(`创建对象池 ${type}, 预分配 ${pool.prealloc} 个对象`);

        return pool;
    }

    // 从池中获取对象
    acquire(type) {
        const pool = this.pools.get(type);
        if (!pool) {
            throw new Error(`对象池 ${type} 不存在`);
        }

        let obj;

        if (pool.available.length > 0) {
            obj = pool.available.pop();
        } else if (pool.inUse.size < pool.maxSize) {
            obj = pool.factory();
        } else {
            // 池已满，强制回收最老的对象
            const oldestObj = pool.inUse.values().next().value;
            this.release(type, oldestObj);
            obj = pool.factory();
        }

        pool.inUse.add(obj);
        obj._poolType = type;
        obj._acquisitionTime = Date.now();

        return obj;
    }

    // 释放对象回池
    release(type, obj) {
        const pool = this.pools.get(type);
        if (!pool || !pool.inUse.has(obj)) {
            return;
        }

        pool.inUse.delete(obj);

        // 重置对象状态
        if (pool.resetFunction) {
            pool.resetFunction(obj);
        }

        pool.available.push(obj);
        delete obj._poolType;
        delete obj._acquisitionTime;
    }

    // 内存监控
    setupMemoryMonitoring() {
        setInterval(() => {
            this.updateMemoryUsage();

            if (this.currentUsage / this.memoryBudget > this.gcThreshold) {
                console.log('内存使用率过高，触发垃圾回收');
                this.performGarbageCollection();
            }
        }, 1000);
    }

    updateMemoryUsage() {
        if (performance.memory) {
            this.currentUsage = performance.memory.usedJSHeapSize;
        }

        // 计算各个池的使用情况
        let poolMemory = 0;
        for (const [type, pool] of this.pools.entries()) {
            const inUseCount = pool.inUse.size;
            const availableCount = pool.available.length;
            poolMemory += (inUseCount + availableCount) * 1000; // 估算每个对象 1KB

            console.log(`池 ${type}: 使用中 ${inUseCount}, 可用 ${availableCount}`);
        }

        console.log(`内存使用: JS堆 ${(this.currentUsage / 1024 / 1024).toFixed(1)}MB, 对象池 ${(poolMemory / 1024 / 1024).toFixed(1)}MB`);
    }

    performGarbageCollection() {
        // 清理长时间未使用的对象
        const maxAge = 30000; // 30秒
        const currentTime = Date.now();

        for (const [type, pool] of this.pools.entries()) {
            const objectsToRelease = [];

            for (const obj of pool.inUse) {
                if (currentTime - obj._acquisitionTime > maxAge) {
                    objectsToRelease.push(obj);
                }
            }

            objectsToRelease.forEach(obj => {
                this.release(type, obj);
            });

            // 清理多余的可用对象
            if (pool.available.length > pool.prealloc * 2) {
                const excess = pool.available.length - pool.prealloc;
                pool.available.splice(0, excess);
            }
        }

        // 强制 JavaScript GC
        if (window.gc) {
            window.gc();
        }
    }

    // 资源特定的池
    createTexturePool() {
        return this.createPool('texture', () => {
            // 创建临时纹理对象
            return {
                texture: null,
                width: 0,
                height: 0,
                format: null,
                inUse: false
            };
        }, {
            maxSize: 50,
            prealloc: 5,
            resetFunction: (obj) => {
                if (obj.texture) {
                    // 注意：不删除纹理，只是标记为可重用
                    obj.inUse = false;
                }
            }
        });
    }

    createMaterialPool() {
        return this.createPool('material', () => {
            return {
                materialInstance: null,
                parameters: new Map(),
                inUse: false
            };
        }, {
            maxSize: 30,
            prealloc: 3,
            resetFunction: (obj) => {
                obj.parameters.clear();
                obj.inUse = false;
            }
        });
    }

// 智能缓存系统
class SmartCacheSystem {
    constructor(engine) {
        this.engine = engine;
        this.caches = new Map();
        this.statistics = new Map();
        this.cleanupInterval = null;

        this.setupCacheTypes();
        this.startPeriodicCleanup();
    }

    setupCacheTypes() {
        // 纹理缓存
        this.createCache('texture', {
            maxSize: 100,
            maxMemory: 200 * 1024 * 1024, // 200MB
            ttl: 300000, // 5分钟
            priority: 'lru'
        });

        // 几何缓存
        this.createCache('geometry', {
            maxSize: 50,
            maxMemory: 100 * 1024 * 1024, // 100MB
            ttl: 600000, // 10分钟
            priority: 'lfu'
        });

        // 材质缓存
        this.createCache('material', {
            maxSize: 200,
            maxMemory: 50 * 1024 * 1024, // 50MB
            ttl: 900000, // 15分钟
            priority: 'lru'
        });
    }

    createCache(type, options) {
        const cache = {
            type: type,
            items: new Map(),
            accessTimes: new Map(),
            frequencies: new Map(),
            sizes: new Map(),
            maxSize: options.maxSize,
            maxMemory: options.maxMemory,
            ttl: options.ttl,
            priority: options.priority,
            currentMemory: 0
        };

        this.caches.set(type, cache);
        this.statistics.set(type, {
            hits: 0,
            misses: 0,
            evictions: 0,
            memoryPressureEvictions: 0
        });
    }

    // 获取缓存项
    get(type, key) {
        const cache = this.caches.get(type);
        const stats = this.statistics.get(type);

        if (!cache || !cache.items.has(key)) {
            stats.misses++;
            return null;
        }

        const item = cache.items.get(key);
        const now = Date.now();

        // 检查 TTL
        if (now - item.timestamp > cache.ttl) {
            this.remove(type, key);
            stats.misses++;
            return null;
        }

        // 更新访问统计
        cache.accessTimes.set(key, now);
        cache.frequencies.set(key, (cache.frequencies.get(key) || 0) + 1);

        stats.hits++;
        return item.data;
    }

    // 设置缓存项
    set(type, key, data, size) {
        const cache = this.caches.get(type);
        if (!cache) return;

        // 检查内存限制
        if (cache.currentMemory + size > cache.maxMemory) {
            this.evictForMemory(type, size);
        }

        // 检查数量限制
        if (cache.items.size >= cache.maxSize) {
            this.evictByPriority(type, 1);
        }

        const now = Date.now();
        const item = {
            data: data,
            size: size,
            timestamp: now
        };

        cache.items.set(key, item);
        cache.sizes.set(key, size);
        cache.accessTimes.set(key, now);
        cache.frequencies.set(key, 1);
        cache.currentMemory += size;
    }

    // 基于内存压力的驱逐
    evictForMemory(type, requiredSize) {
        const cache = this.caches.get(type);
        const stats = this.statistics.get(type);
        let freedMemory = 0;

        const itemsToEvict = this.selectEvictionCandidates(type);

        for (const key of itemsToEvict) {
            if (freedMemory >= requiredSize) break;

            const size = cache.sizes.get(key);
            this.remove(type, key);
            freedMemory += size;
            stats.memoryPressureEvictions++;
        }
    }

    // 基于优先级的驱逐
    evictByPriority(type, count) {
        const cache = this.caches.get(type);
        const stats = this.statistics.get(type);

        const candidates = this.selectEvictionCandidates(type);

        for (let i = 0; i < count && i < candidates.length; i++) {
            this.remove(type, candidates[i]);
            stats.evictions++;
        }
    }

    selectEvictionCandidates(type) {
        const cache = this.caches.get(type);
        const items = Array.from(cache.items.keys());

        switch (cache.priority) {
            case 'lru':
                return items.sort((a, b) =>
                    cache.accessTimes.get(a) - cache.accessTimes.get(b));

            case 'lfu':
                return items.sort((a, b) =>
                    cache.frequencies.get(a) - cache.frequencies.get(b));

            case 'ttl':
                return items.sort((a, b) => {
                    const ageA = Date.now() - cache.items.get(a).timestamp;
                    const ageB = Date.now() - cache.items.get(b).timestamp;
                    return ageB - ageA;
                });

            default:
                return items;
        }
    }

    remove(type, key) {
        const cache = this.caches.get(type);
        if (!cache || !cache.items.has(key)) return;

        const size = cache.sizes.get(key);
        const item = cache.items.get(key);

        // 清理资源
        if (item.data && typeof item.data.delete === 'function') {
            item.data.delete();
        }

        cache.items.delete(key);
        cache.sizes.delete(key);
        cache.accessTimes.delete(key);
        cache.frequencies.delete(key);
        cache.currentMemory -= size;
    }

    // 定期清理
    startPeriodicCleanup() {
        this.cleanupInterval = setInterval(() => {
            this.performPeriodicCleanup();
        }, 60000); // 每分钟清理一次
    }

    performPeriodicCleanup() {
        const now = Date.now();

        for (const [type, cache] of this.caches.entries()) {
            const expiredKeys = [];

            for (const [key, item] of cache.items.entries()) {
                if (now - item.timestamp > cache.ttl) {
                    expiredKeys.push(key);
                }
            }

            expiredKeys.forEach(key => this.remove(type, key));

            if (expiredKeys.length > 0) {
                console.log(`清理了 ${expiredKeys.length} 个过期的 ${type} 缓存项`);
            }
        }
    }

    // 获取缓存统计
    getStatistics() {
        const allStats = {};

        for (const [type, cache] of this.caches.entries()) {
            const stats = this.statistics.get(type);
            const hitRate = stats.hits / (stats.hits + stats.misses) * 100;

            allStats[type] = {
                items: cache.items.size,
                memory: `${(cache.currentMemory / 1024 / 1024).toFixed(1)}MB`,
                hitRate: `${hitRate.toFixed(1)}%`,
                hits: stats.hits,
                misses: stats.misses,
                evictions: stats.evictions
            };
        }

        return allStats;
    }

    destroy() {
        if (this.cleanupInterval) {
            clearInterval(this.cleanupInterval);
        }

        for (const [type, cache] of this.caches.entries()) {
            for (const key of cache.items.keys()) {
                this.remove(type, key);
            }
        }
    }
}
```

```javascript
// 智能资源加载器
class SmartResourceLoader {
    constructor() {
        this.loadingQueue = [];
        this.activeRequests = new Map();
        this.cache = new Map();
        this.preloadedResources = new Set();
        this.maxConcurrentRequests = 6;
        this.bandwidthMonitor = new BandwidthMonitor();

        this.setupRequestInterceptor();
    }

    // 预加载策略
    async preloadCriticalResources(manifest) {
        console.log('开始预加载关键资源...');

        const criticalResources = manifest.filter(item => item.priority === 'critical');
        const importantResources = manifest.filter(item => item.priority === 'important');

        // 并行加载关键资源
        await this.loadResourceBatch(criticalResources, 'critical');

        // 后台加载重要资源
        this.loadResourceBatch(importantResources, 'background');

        console.log('关键资源预加载完成');
    }

    async loadResourceBatch(resources, mode = 'normal') {
        const concurrency = mode === 'critical' ? this.maxConcurrentRequests : 2;

        const batches = this.createBatches(resources, concurrency);

        for (const batch of batches) {
            const promises = batch.map(resource => this.loadResource(resource));

            if (mode === 'critical') {
                await Promise.all(promises);
            } else {
                // 后台模式：不阻塞，但限制并发
                Promise.all(promises).catch(error => {
                    console.warn('后台资源加载失败:', error);
                });

                // 给主线程让出时间
                await this.delay(50);
            }
        }
    }

    createBatches(items, batchSize) {
        const batches = [];
        for (let i = 0; i < items.length; i += batchSize) {
            batches.push(items.slice(i, i + batchSize));
        }
        return batches;
    }

    // 自适应质量加载
    async loadWithAdaptiveQuality(url, qualityLevels) {
        const networkSpeed = this.bandwidthMonitor.getCurrentSpeed();
        const deviceCapability = this.assessDeviceCapability();

        // 选择合适的质量级别
        const selectedQuality = this.selectQualityLevel(
            networkSpeed, deviceCapability, qualityLevels
        );

        console.log(`选择质量级别: ${selectedQuality.name} (带宽: ${networkSpeed}Mbps)`);

        try {
            // 加载选定质量的资源
            const resource = await this.loadResource(selectedQuality);

            // 如果网络条件良好，异步加载更高质量版本
            if (selectedQuality.name !== 'high' && networkSpeed > 10) {
                this.upgradeResourceQuality(url, qualityLevels, selectedQuality);
            }

            return resource;
        } catch (error) {
            // 失败时降级到更低质量
            return this.fallbackToLowerQuality(url, qualityLevels, selectedQuality);
        }
    }

    selectQualityLevel(networkSpeed, deviceCapability, levels) {
        // 基于网络速度和设备能力选择质量
        let selectedLevel = levels.find(level => level.name === 'low');

        if (networkSpeed > 5 && deviceCapability !== 'low') {
            selectedLevel = levels.find(level => level.name === 'medium') || selectedLevel;
        }

        if (networkSpeed > 15 && deviceCapability === 'high') {
            selectedLevel = levels.find(level => level.name === 'high') || selectedLevel;
        }

        return selectedLevel;
    }

    // 渐进式加载
    async loadProgressively(baseUrl, progressLevels, onLevelLoaded) {
        const progressiveLoader = {
            currentLevel: 0,
            levels: progressLevels,
            resources: new Map()
        };

        // 加载第一级（最低质量）
        const firstLevel = progressLevels[0];
        const initialResource = await this.loadResource(firstLevel);
        progressiveLoader.resources.set(0, initialResource);

        onLevelLoaded(0, initialResource);

        // 后台加载更高级别
        this.loadHigherLevelsInBackground(progressiveLoader, onLevelLoaded);

        return progressiveLoader;
    }

    async loadHigherLevelsInBackground(progressiveLoader, onLevelLoaded) {
        for (let i = 1; i < progressiveLoader.levels.length; i++) {
            // 检查网络条件
            if (this.bandwidthMonitor.getCurrentSpeed() < 2) {
                console.log('网络速度较慢，暂停高质量加载');
                break;
            }

            try {
                const level = progressiveLoader.levels[i];
                const resource = await this.loadResource(level);

                progressiveLoader.resources.set(i, resource);
                progressiveLoader.currentLevel = i;

                onLevelLoaded(i, resource);

                console.log(`加载完成级别 ${i}: ${level.name}`);

                // 控制加载速度，避免影响主线程
                await this.delay(100);

            } catch (error) {
                console.warn(`级别 ${i} 加载失败:`, error);
                break;
            }
        }
    }

    // 请求拦截和优化
    setupRequestInterceptor() {
        const originalFetch = window.fetch;

        window.fetch = async (url, options = {}) => {
            // 检查缓存
            if (this.cache.has(url)) {
                const cachedResponse = this.cache.get(url);
                if (this.isCacheValid(cachedResponse)) {
                    return cachedResponse.response.clone();
                }
            }

            // 请求去重
            if (this.activeRequests.has(url)) {
                return this.activeRequests.get(url);
            }

            // 优化请求头
            const optimizedOptions = this.optimizeRequestOptions(options);

            const requestPromise = originalFetch(url, optimizedOptions)
                .then(response => {
                    // 缓存成功响应
                    if (response.ok) {
                        this.cache.set(url, {
                            response: response.clone(),
                            timestamp: Date.now(),
                            ttl: this.calculateTTL(url)
                        });
                    }

                    this.activeRequests.delete(url);
                    return response;
                })
                .catch(error => {
                    this.activeRequests.delete(url);
                    throw error;
                });

            this.activeRequests.set(url, requestPromise);
            return requestPromise;
        };
    }

    optimizeRequestOptions(options) {
        const optimized = { ...options };

        // 添加压缩支持
        optimized.headers = {
            'Accept-Encoding': 'br, gzip, deflate',
            ...optimized.headers
        };

        // 设置缓存策略
        if (!optimized.cache) {
            optimized.cache = 'default';
        }

        // 添加优先级提示
        if (optimized.priority) {
            optimized.headers['Priority'] = optimized.priority;
        }

        return optimized;
    }

    calculateTTL(url) {
        // 基于文件类型设置缓存时间
        if (url.includes('.wasm') || url.includes('.js')) {
            return 3600000; // 1小时
        } else if (url.includes('.gltf') || url.includes('.glb')) {
            return 1800000; // 30分钟
        } else if (url.includes('.ktx') || url.includes('.png') || url.includes('.jpg')) {
            return 7200000; // 2小时
        }

        return 600000; // 默认10分钟
    }

    delay(ms) {
        return new Promise(resolve => setTimeout(resolve, ms));
    }
}

class BandwidthMonitor {
    constructor() {
        this.measurements = [];
        this.isMonitoring = false;
        this.currentSpeed = 5; // 默认 5Mbps

        this.startMonitoring();
    }

    startMonitoring() {
        if (this.isMonitoring) return;

        this.isMonitoring = true;
        this.performBandwidthTest();

        // 定期测试带宽
        setInterval(() => {
            this.performBandwidthTest();
        }, 30000); // 每30秒测试一次
    }

    async performBandwidthTest() {
        try {
            // 使用小文件测试带宽
            const testUrl = this.generateTestUrl();
            const startTime = performance.now();

            const response = await fetch(testUrl);
            const data = await response.arrayBuffer();

            const endTime = performance.now();
            const duration = (endTime - startTime) / 1000; // 秒
            const bytes = data.byteLength;
            const bitsPerSecond = (bytes * 8) / duration;
            const mbps = bitsPerSecond / (1024 * 1024);

            this.measurements.push({
                timestamp: Date.now(),
                speed: mbps,
                bytes: bytes,
                duration: duration
            });

            // 保持最近20个测量值
            if (this.measurements.length > 20) {
                this.measurements.shift();
            }

            // 计算平均速度
            this.currentSpeed = this.calculateAverageSpeed();

            console.log(`带宽测试: ${mbps.toFixed(2)} Mbps (平均: ${this.currentSpeed.toFixed(2)} Mbps)`);

        } catch (error) {
            console.warn('带宽测试失败:', error);
        }
    }

    generateTestUrl() {
        // 生成随机查询参数避免缓存
        const timestamp = Date.now();
        const random = Math.random();

        // 使用一个小的测试文件（如 10KB 图片）
        return `test-bandwidth.jpg?t=${timestamp}&r=${random}`;
    }

    calculateAverageSpeed() {
        if (this.measurements.length === 0) return this.currentSpeed;

        // 计算最近5次测量的平均值
        const recent = this.measurements.slice(-5);
        const sum = recent.reduce((total, measurement) => total + measurement.speed, 0);

        return sum / recent.length;
    }

    getCurrentSpeed() {
        return this.currentSpeed;
    }

    getSpeedCategory() {
        if (this.currentSpeed < 1) return 'very_slow';
        if (this.currentSpeed < 5) return 'slow';
        if (this.currentSpeed < 15) return 'medium';
        if (this.currentSpeed < 50) return 'fast';
        return 'very_fast';
    }
```

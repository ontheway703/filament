```objc
#import "FilamentView.h"

// Filament 头文件
#include <filament/Engine.h>
#include <filament/Renderer.h>
#include <filament/Scene.h>
#include <filament/View.h>
#include <filament/Camera.h>
#include <filament/SwapChain.h>

// glTF 支持
#include <gltfio/AssetLoader.h>
#include <gltfio/FilamentAsset.h>

// 相机工具
#include <camutils/Manipulator.h>

// 数学库
#include <math/vec3.h>
#include <math/mat4.h>

using namespace filament;
using namespace gltfio;
using namespace camutils;

@interface FilamentView () {
    // Filament 核心对象
    Engine* engine;
    Renderer* renderer;
    Scene* scene;
    View* filaView;
    Camera* camera;
    SwapChain* swapChain;

    // 资产管理
    AssetLoader* assetLoader;
    FilamentAsset* currentAsset;

    // 相机控制
    Manipulator* cameraController;

    // 渲染循环
    CADisplayLink* displayLink;

    // 状态管理
    BOOL isRenderingActive;
}

@property (nonatomic, strong) CAMetalLayer* metalLayer;

@end

@implementation FilamentView

#pragma mark - 初始化

+ (Class)layerClass {
    return [CAMetalLayer class];
}

- (instancetype)initWithFrame:(CGRect)frame {
    self = [super initWithFrame:frame];
    if (self) {
        [self setupMetalLayer];
        [self setupFilament];
        [self setupGestures];
        [self setupNotifications];
    }
    return self;
}

- (void)setupMetalLayer {
    self.metalLayer = (CAMetalLayer*)self.layer;
    self.metalLayer.device = MTLCreateSystemDefaultDevice();
    self.metalLayer.pixelFormat = MTLPixelFormatBGRA8Unorm;
    self.metalLayer.framebufferOnly = YES;
}

- (void)setupFilament {
    // 创建引擎（优先使用 Metal）
    engine = Engine::create(Engine::Backend::METAL);
    if (!engine) {
        NSLog(@"Failed to create Metal engine, falling back to OpenGL");
        engine = Engine::create(Engine::Backend::OPENGL);
    }

    // 创建基础组件
    renderer = engine->createRenderer();
    scene = engine->createScene();
    filaView = engine->createView();

    // 设置视图场景关联
    filaView->setScene(scene);

    // 创建相机
    utils::Entity cameraEntity = utils::EntityManager::get().create();
    camera = engine->createCamera(cameraEntity);
    filaView->setCamera(camera);

    // 创建交换链
    swapChain = engine->createSwapChain((__bridge void*)self.metalLayer);

    // 创建资产加载器
    assetLoader = AssetLoader::create({
        .engine = engine,
        .materials = createJitShaderProvider(engine)
    });

    // 设置默认相机控制器
    [self setupCameraController];

    // 配置默认渲染设置
    [self setupDefaultRenderSettings];
}

@end
```

```objc
// SceneModel.h
@interface SceneModel : NSObject

@property (nonatomic, readonly) NSArray<NSURL*>* availableModels;
@property (nonatomic, strong) NSURL* currentModelURL;
@property (nonatomic, assign) float lightIntensity;
@property (nonatomic, assign) simd_float3 lightDirection;

- (void)loadModelAtIndex:(NSUInteger)index completion:(void(^)(BOOL success))completion;
- (void)updateLighting;

@end

// SceneModel.m
@implementation SceneModel

- (instancetype)init {
    self = [super init];
    if (self) {
        _lightIntensity = 100000.0f;
        _lightDirection = simd_make_float3(0.0f, -1.0f, 0.0f);
        [self loadAvailableModels];
    }
    return self;
}

- (void)loadAvailableModels {
    NSMutableArray* models = [NSMutableArray array];

    // 扫描 Bundle 中的 glTF 文件
    NSArray* extensions = @[@"glb", @"gltf"];
    for (NSString* ext in extensions) {
        NSArray* paths = [[NSBundle mainBundle] pathsForResourcesOfType:ext inDirectory:nil];
        for (NSString* path in paths) {
            [models addObject:[NSURL fileURLWithPath:path]];
        }
    }

    _availableModels = [models copy];
}

@end
```

```objc
// FilamentViewController.h
@interface FilamentViewController : UIViewController

@property (nonatomic, strong) SceneModel* sceneModel;
@property (nonatomic, weak) IBOutlet FilamentView* filamentView;
@property (nonatomic, weak) IBOutlet UISlider* lightIntensitySlider;
@property (nonatomic, weak) IBOutlet UISegmentedControl* modelSelector;

@end

// FilamentViewController.m
@implementation FilamentViewController

- (void)viewDidLoad {
    [super viewDidLoad];

    self.sceneModel = [[SceneModel alloc] init];
    [self setupUI];
    [self bindModelToView];
}

- (void)setupUI {
    // 配置模型选择器
    [self.modelSelector removeAllSegments];
    for (NSUInteger i = 0; i < self.sceneModel.availableModels.count; i++) {
        NSURL* modelURL = self.sceneModel.availableModels[i];
        NSString* name = [modelURL lastPathComponent];
        [self.modelSelector insertSegmentWithTitle:name atIndex:i animated:NO];
    }

    if (self.modelSelector.numberOfSegments > 0) {
        self.modelSelector.selectedSegmentIndex = 0;
    }

    // 配置光照滑块
    self.lightIntensitySlider.minimumValue = 1000.0f;
    self.lightIntensitySlider.maximumValue = 200000.0f;
    self.lightIntensitySlider.value = self.sceneModel.lightIntensity;
}

- (IBAction)modelChanged:(UISegmentedControl*)sender {
    [self.sceneModel loadModelAtIndex:sender.selectedSegmentIndex
                           completion:^(BOOL success) {
        if (success) {
            [self.filamentView loadCurrentModel];
        }
    }];
}

- (IBAction)lightIntensityChanged:(UISlider*)sender {
    self.sceneModel.lightIntensity = sender.value;
    [self.sceneModel updateLighting];
    [self.filamentView updateLighting];
}

@end
```

```objc
    static AssetLoadingManager* instance = nil;
    static dispatch_once_t onceToken;

    return instance;

- (void)loadAssetFromURL:(NSURL*)url completion:(void(^)(FilamentAsset*, NSError*))completion {
    dispatch_async(dispatch_get_global_queue(DISPATCH_QUEUE_PRIORITY_DEFAULT, 0), ^{
        @autoreleasepool {
            NSError* error = nil;
            NSData* data = [NSData dataWithContentsOfURL:url options:0 error:&error];

            if (!data || error) {

                    completion(nil, error);

                completion(asset, nil);
            });
```

### 3. 性能监控和优化

#### 性能统计组件
```objc
@interface PerformanceMonitor : NSObject

@property (nonatomic, readonly) float averageFPS;
@property (nonatomic, readonly) float averageFrameTime;
@property (nonatomic, readonly) NSUInteger triangleCount;
@property (nonatomic, readonly) NSUInteger drawCallCount;

- (void)beginFrame;
- (void)endFrame;
- (void)recordDrawCall:(NSUInteger)triangles;
- (void)reset;

@end

@implementation PerformanceMonitor {
    CFTimeInterval frameStartTime;
    NSMutableArray<NSNumber*>* frameTimes;
    NSUInteger currentTriangleCount;
    NSUInteger currentDrawCallCount;
    NSUInteger frameCounter;
}

- (instancetype)init {
    self = [super init];
    if (self) {
        frameTimes = [NSMutableArray arrayWithCapacity:60]; // 保持60帧数据
        [self reset];
    }
    return self;
}

- (void)beginFrame {
    frameStartTime = CACurrentMediaTime();
    currentTriangleCount = 0;
    currentDrawCallCount = 0;
}

- (void)endFrame {
    CFTimeInterval frameTime = CACurrentMediaTime() - frameStartTime;

    [frameTimes addObject:@(frameTime)];
    if (frameTimes.count > 60) {
        [frameTimes removeObjectAtIndex:0];
    }

    frameCounter++;

    // 每秒输出一次统计
    if (frameCounter % 60 == 0) {
        [self logPerformanceStats];
    }
}

- (void)recordDrawCall:(NSUInteger)triangles {
    currentDrawCallCount++;
    currentTriangleCount += triangles;
}

- (float)averageFrameTime {
    if (frameTimes.count == 0) return 0.0f;

    double total = 0.0;
    for (NSNumber* time in frameTimes) {
        total += time.doubleValue;
    }
    return (float)(total / frameTimes.count);
}

- (float)averageFPS {
    float frameTime = self.averageFrameTime;
    return frameTime > 0 ? (1.0f / frameTime) : 0.0f;
}

- (void)logPerformanceStats {
    NSLog(@"Performance Stats - FPS: %.1f, Frame Time: %.2fms, Draw Calls: %lu, Triangles: %lu",
          self.averageFPS, self.averageFrameTime * 1000.0f,
          (unsigned long)currentDrawCallCount, (unsigned long)currentTriangleCount);
}

@end
```

### 4. 内存管理最佳实践

#### 内存监控器
```objc
@interface MemoryManager : NSObject

+ (instancetype)sharedManager;
- (void)trackAllocation:(size_t)size forCategory:(NSString*)category;
- (void)trackDeallocation:(size_t)size forCategory:(NSString*)category;
- (void)logMemoryUsage;
- (void)handleMemoryWarning;

@end

@implementation MemoryManager {
    NSMutableDictionary<NSString*, NSNumber*>* categoryUsage;
    dispatch_queue_t trackingQueue;
}

+ (instancetype)sharedManager {
    static MemoryManager* instance = nil;
    static dispatch_once_t onceToken;
    dispatch_once(&onceToken, ^{
        instance = [[MemoryManager alloc] init];
    });
    return instance;
}

- (instancetype)init {
    self = [super init];
    if (self) {
        categoryUsage = [NSMutableDictionary dictionary];
        trackingQueue = dispatch_queue_create("com.filament.memory.tracking", DISPATCH_QUEUE_SERIAL);

        // 监听内存警告
        [[NSNotificationCenter defaultCenter] addObserver:self
                                                 selector:@selector(handleMemoryWarning)
                                                     name:UIApplicationDidReceiveMemoryWarningNotification
                                                   object:nil];
    }
    return self;
}

- (void)trackAllocation:(size_t)size forCategory:(NSString*)category {
    dispatch_async(trackingQueue, ^{
        NSNumber* currentUsage = self->categoryUsage[category] ?: @0;
        self->categoryUsage[category] = @(currentUsage.unsignedLongValue + size);
    });
}

- (void)trackDeallocation:(size_t)size forCategory:(NSString*)category {
    dispatch_async(trackingQueue, ^{
        NSNumber* currentUsage = self->categoryUsage[category] ?: @0;
        unsigned long newUsage = currentUsage.unsignedLongValue;
        newUsage = (newUsage > size) ? (newUsage - size) : 0;
        self->categoryUsage[category] = @(newUsage);
    });
}

- (void)handleMemoryWarning {
    NSLog(@"Memory warning received, current usage:");
    [self logMemoryUsage];

    // 通知应用清理资源
    [[NSNotificationCenter defaultCenter] postNotificationName:@"FilamentMemoryWarning" object:nil];
}

@end
```

```objc
    // 检查相机设置
    math::mat4f projection = camera->getProjectionMatrix();
    NSAssert(!isnan(projection[0][0]), @"Invalid camera projection");

    FIL_LOG_DEBUG(@"Render state validation passed");
}
```

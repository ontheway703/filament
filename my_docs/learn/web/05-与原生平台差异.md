# Filament Web 与原生平台差异

## 1. 平台架构对比

### 1.1 整体架构差异

**原生平台架构**
```
应用层 (C++/Java/Swift)
        ↓
Filament Engine (C++)
        ↓
Backend (Vulkan/Metal/OpenGL)
        ↓
操作系统图形驱动
        ↓
GPU 硬件
```

**Web 平台架构**
```
JavaScript 应用层
        ↓
Embind 绑定层
        ↓
WebAssembly 模块
        ↓
WebGL 2.0 API
        ↓
浏览器图形抽象层
        ↓
操作系统图形驱动
        ↓
GPU 硬件
```

### 1.2 关键差异点

| 特性 | 原生平台 | Web 平台 | 影响 |
|------|----------|----------|------|
| 执行环境 | 原生机器码 | WebAssembly | 性能损失 10-30% |
| 图形 API | Vulkan/Metal/OpenGL | WebGL 2.0 | 功能受限 |
| 内存管理 | 直接内存访问 | 沙箱内存模型 | 额外内存拷贝 |
| 多线程 | 完整支持 | 受限支持 | 并发能力降低 |
| 文件 I/O | 直接文件系统 | 虚拟文件系统 | 异步加载必需 |

```javascript
            this.renderer.render(this.view);
```

### 4.2 Web 平台单线程限制

```javascript
// Web 平台的单线程解决方案
class WebWorkerSolution {
    constructor() {
        this.workers = [];
        this.taskQueue = [];
        this.setupWorkers();
    }

    setupWorkers() {
        // 创建 Web Workers 处理计算密集型任务
        const workerCount = navigator.hardwareConcurrency || 4;

        for (let i = 0; i < workerCount; i++) {
            const worker = new Worker('asset-processor-worker.js');
            worker.onmessage = this.handleWorkerMessage.bind(this);
            this.workers.push(worker);
        }
    }

    // 将资源处理任务分发到 Web Workers
    processAssetAsync(assetData, options) {
        return new Promise((resolve, reject) => {
            const availableWorker = this.findAvailableWorker();

            if (availableWorker) {
                const taskId = Date.now() + Math.random();

                this.taskQueue.push({
                    id: taskId,
                    resolve,
                    reject
                });

                availableWorker.postMessage({
                    taskId,
                    assetData,
                    options,
                    type: 'PROCESS_ASSET'
                });
            } else {
                // 如果没有可用的 Worker，降级到主线程处理
                this.processAssetOnMainThread(assetData, options)
                    .then(resolve)
                    .catch(reject);
            }
        });
    }

    handleWorkerMessage(event) {
        const { taskId, result, error } = event.data;

        const taskIndex = this.taskQueue.findIndex(task => task.id === taskId);
        if (taskIndex !== -1) {
            const task = this.taskQueue[taskIndex];
            this.taskQueue.splice(taskIndex, 1);

            if (error) {
                task.reject(new Error(error));
            } else {
                task.resolve(result);
            }
        }
    }

    // 主线程分片处理
    async processAssetOnMainThread(assetData, options) {
        const CHUNK_SIZE = 1000; // 每次处理1000个顶点
        const vertices = assetData.vertices;
        const processedVertices = [];

        for (let i = 0; i < vertices.length; i += CHUNK_SIZE) {
            const chunk = vertices.slice(i, i + CHUNK_SIZE);
            const processedChunk = await this.processVertexChunk(chunk, options);
            processedVertices.push(...processedChunk);

            // 让出控制权，避免阻塞渲染
            await this.yieldControl();
        }

        return { vertices: processedVertices };
    }

    yieldControl() {
        return new Promise(resolve => {
            setTimeout(resolve, 0);
        });
    }

    findAvailableWorker() {
        // 简化实现：返回第一个 Worker
        // 实际中应该跟踪 Worker 状态
        return this.workers[0];
    }
}
```

```cpp
// 原生平台直接文件系统访问
class NativeAssetLoader {
public:
    std::vector<uint8_t> loadFile(const std::string& filepath) {
        std::ifstream file(filepath, std::ios::binary | std::ios::ate);

        if (!file) {
            throw std::runtime_error("无法打开文件: " + filepath);
        }

        std::streamsize size = file.tellg();
        file.seekg(0, std::ios::beg);

        std::vector<uint8_t> buffer(size);
        file.read(reinterpret_cast<char*>(buffer.data()), size);

        return buffer;
    }

    // 同步加载多个文件
    std::vector<Asset> loadAssets(const std::vector<std::string>& filepaths) {
        std::vector<Asset> assets;

        for (const auto& path : filepaths) {
            auto data = loadFile(path);
            assets.emplace_back(processAssetData(data));
        }

        return assets;
    }

    // 异步批量加载
    std::future<std::vector<Asset>> loadAssetsAsync(
        const std::vector<std::string>& filepaths) {

        return std::async(std::launch::async, [this, filepaths]() {
            return loadAssets(filepaths);
        });
    }
};
```

```javascript
// Web 平台虚拟文件系统和异步加载
class WebAssetLoader {
    constructor() {
        this.cache = new Map();
        this.loadingPromises = new Map();
        this.virtualFS = new Map();
    }

    // 预加载资源到虚拟文件系统
    async preloadAssets(assetManifest) {
        const loadPromises = assetManifest.map(async (asset) => {
            try {
                const response = await fetch(asset.url);
                const arrayBuffer = await response.arrayBuffer();

                // 存储到虚拟文件系统
                this.virtualFS.set(asset.name, {
                    data: new Uint8Array(arrayBuffer),
                    metadata: asset.metadata,
                    loadTime: Date.now()
                });

                return { name: asset.name, status: 'success' };
            } catch (error) {
                return { name: asset.name, status: 'error', error };
            }
        });

        const results = await Promise.allSettled(loadPromises);
        return results;
    }

    // 从虚拟文件系统读取
    readFile(filename) {
        const file = this.virtualFS.get(filename);
        if (!file) {
            throw new Error(`文件不存在: ${filename}`);
        }

        return file.data;
    }

    // 异步网络加载 (带重试)
    async loadFromNetwork(url, retries = 3) {
        for (let attempt = 0; attempt < retries; attempt++) {
            try {
                const response = await fetch(url, {
                    cache: 'default',
                    mode: 'cors'
                });

                if (!response.ok) {
                    throw new Error(`HTTP ${response.status}: ${response.statusText}`);
                }

                return await response.arrayBuffer();
            } catch (error) {
                console.warn(`加载失败 (${attempt + 1}/${retries}):`, error);

                if (attempt === retries - 1) {
                    throw error;
                }

                // 指数退避重试
                await this.delay(Math.pow(2, attempt) * 1000);
            }
        }
    }

    // 流式加载大文件
    async loadLargeFileStream(url, onProgress) {
        const response = await fetch(url);
        const reader = response.body.getReader();
        const contentLength = +response.headers.get('Content-Length');

        let receivedLength = 0;
        const chunks = [];

        while (true) {
            const { done, value } = await reader.read();

            if (done) break;

            chunks.push(value);
            receivedLength += value.length;

            if (onProgress && contentLength) {
                onProgress(receivedLength / contentLength);
            }
        }

        // 合并所有块
        const allChunks = new Uint8Array(receivedLength);
        let position = 0;

        for (const chunk of chunks) {
            allChunks.set(chunk, position);
            position += chunk.length;
        }

        return allChunks.buffer;
    }

    delay(ms) {
        return new Promise(resolve => setTimeout(resolve, ms));
    }

    // 资源缓存策略
    setCacheStrategy(strategy) {
        this.cacheStrategy = strategy;
    }

    shouldCache(url, size) {
        switch (this.cacheStrategy) {
            case 'aggressive':
                return size < 10 * 1024 * 1024; // 10MB以下

            case 'conservative':
                return size < 1024 * 1024; // 1MB以下

            case 'none':
                return false;

            default:
                return size < 5 * 1024 * 1024; // 5MB以下
        }
    }
}
```

```javascript
            // 减少变量数量以适应限制
            uniform vec3 lightDirection;
            uniform vec3 lightColor;

            in vec2 texCoord;
            in vec3 worldNormal;
            in vec3 worldPosition;

            out vec4 fragColor;

            void main() {
                vec3 albedo = texture(albedoMap, texCoord).rgb;
                vec3 normal = normalize(worldNormal);
                float roughness = texture(roughnessMap, texCoord).r;

                // 简化的光照计算
                float NdotL = max(dot(normal, -lightDirection), 0.0);
                vec3 diffuse = albedo * lightColor * NdotL;

                fragColor = vec4(diffuse, 1.0);
            }
        `;

        return this.compileMaterial(engine, shaderCode);
    }

    createFullFeatureMaterial(engine) {
        // 原生平台完整功能版本
        const shaderFeatures = {
            // 高级光照
            imageBasedLighting: true,
            areaLights: true,
            volumetricLighting: true,

            // 高级着色
            subsurfaceScattering: true,
            anisotropicReflection: true,
            clearCoat: true,

            // 几何特效
            parallaxMapping: true,
            displacementMapping: true,

            // 后处理集成
            screenSpaceReflections: true,
            ambientOcclusion: true
        };

        return this.compileAdvancedMaterial(engine, shaderFeatures);
    }
}
```

```javascript
            if (timerExt) {
                this.setupTimerQueries(gl, timerExt);
            } else {
                console.warn('GPU 时间查询不可用');
            }
        }
    }

    setupTimerQueries(gl, timerExt) {
        let query = gl.createQuery();

        // 开始查询
        gl.beginQuery(timerExt.TIME_ELAPSED_EXT, query);

        // 执行渲染
        this.renderFrame();

        gl.endQuery(timerExt.TIME_ELAPSED_EXT);

        // 异步获取结果
        this.checkQueryResult(gl, timerExt, query);
    }

    checkQueryResult(gl, timerExt, query) {
        const checkResult = () => {
            if (gl.getQueryParameter(query, gl.QUERY_RESULT_AVAILABLE)) {
                const timeElapsed = gl.getQueryParameter(query, gl.QUERY_RESULT);
                console.log(`GPU 时间: ${timeElapsed / 1000000}ms`);

                gl.deleteQuery(query);
            } else {
                // 查询仍在进行中
                requestAnimationFrame(checkResult);
            }
        };

        checkResult();
    }
```

## 9. 总结对比表

| 方面 | 原生平台 | Web 平台 | 影响程度 |
|------|----------|----------|----------|
| **性能** | 100% | 70-85% | 高 |
| **图形功能** | 完整支持 | WebGL 2.0 限制 | 高 |
| **多线程** | 完整支持 | 严重受限 | 高 |
| **内存使用** | 高效直接 | 双重开销 | 中 |
| **启动时间** | 即时 | 3-10秒 | 中 |
| **文件 I/O** | 同步直接 | 异步网络 | 中 |
| **调试工具** | 完整 | 受限 | 中 |
| **部署复杂度** | 高 | 低 | 低 |
| **平台覆盖** | 单一 | 跨平台 | 低 |
| **更新机制** | 应用商店 | 自动更新 | 低 |

Web 平台虽然在性能和功能上存在限制，但其跨平台特性、便捷的部署方式和自动更新能力，使其在特定应用场景下具有独特的优势。开发者需要根据具体需求权衡选择合适的平台。

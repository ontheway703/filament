# Filament Web 构建系统与工具链

## 1. 构建系统概览

Filament Web 使用 Emscripten 工具链将 C++ 代码编译为 WebAssembly，配合 CMake 构建系统实现跨平台编译。整个构建流程包含预处理、编译、链接和优化等多个阶段。

### 1.1 工具链组件

```bash
# 主要工具链组件
emcc           # C/C++ 编译器 (基于 Clang)
em++           # C++ 编译器
emlink         # 链接器
emar           # 静态库创建工具
emranlib       # 库索引工具
emcmake        # CMake 包装器
emmake         # Make 包装器
```

### 1.2 编译流程图

```
Source Files (.cpp/.h)
         ↓
    Preprocessor
         ↓
   Clang Frontend
         ↓
     LLVM IR
         ↓
   Emscripten Backend
         ↓
    WebAssembly (.wasm)
         +
   JavaScript Glue (.js)
         ↓
    Final Bundle
```

## 4. 构建脚本实现

### 4.1 主构建脚本

```bash
#!/bin/bash
# build-web.sh - Web 平台构建脚本

set -e

# 配置变量
EMSCRIPTEN_ROOT=${EMSDK}/upstream/emscripten
BUILD_TYPE=${1:-Release}
OUTPUT_DIR="out/web-${BUILD_TYPE,,}"
INSTALL_DIR="dist/web"

# 检查 Emscripten 环境
check_emscripten() {
    if [ -z "$EMSDK" ]; then
        echo "错误: 未设置 EMSDK 环境变量"
        echo "请运行: source /path/to/emsdk/emsdk_env.sh"
        exit 1
    fi

    if ! command -v emcc &> /dev/null; then
        echo "错误: 找不到 emcc 编译器"
        exit 1
    fi

    echo "使用 Emscripten 版本: $(emcc --version | head -n1)"
}

# 配置构建目录
setup_build_dir() {
    echo "设置构建目录: $OUTPUT_DIR"
    mkdir -p "$OUTPUT_DIR"
    cd "$OUTPUT_DIR"
}

# 执行 CMake 配置
configure_cmake() {
    echo "配置 CMake..."
    emcmake cmake \
        -DCMAKE_BUILD_TYPE="$BUILD_TYPE" \
        -DFILAMENT_ENABLE_WEBGL=ON \
        -DFILAMENT_ENABLE_JAVA=OFF \
        -DFILAMENT_ENABLE_LTO=ON \
        -DFILAMENT_SKIP_SAMPLES=ON \
        -DFILAMENT_SUPPORTS_VULKAN=OFF \
        -DFILAMENT_SUPPORTS_METAL=OFF \
        ../../..
}

# 编译项目
build_project() {
    echo "编译项目..."
    emmake make -j$(nproc) filament-js

    if [ $? -eq 0 ]; then
        echo "编译成功!"
    else
        echo "编译失败!"
        exit 1
    fi
}

# 安装文件
install_files() {
    echo "安装文件到: $INSTALL_DIR"
    mkdir -p "../../../$INSTALL_DIR"

    # 复制主要文件
    cp web/filament.js "../../../$INSTALL_DIR/"
    cp web/filament.wasm "../../../$INSTALL_DIR/"

    # 复制类型定义
    if [ -f "../../../web/filament-js/filament.d.ts" ]; then
        cp "../../../web/filament-js/filament.d.ts" "../../../$INSTALL_DIR/"
    fi

    # 复制扩展文件
    cp "../../../web/filament-js/extensions.js" "../../../$INSTALL_DIR/"
    cp "../../../web/filament-js/utilities.js" "../../../$INSTALL_DIR/"

    echo "安装完成!"
}

# 主函数
main() {
    echo "开始构建 Filament Web (${BUILD_TYPE})"

    check_emscripten
    setup_build_dir
    configure_cmake
    build_project
    install_files

    echo "构建完成! 输出目录: $INSTALL_DIR"
}

# 运行主函数
main "$@"
```

### 4.2 开发模式构建

```bash
#!/bin/bash
# build-dev.sh - 开发模式构建脚本

# 快速开发构建配置
EMCC_FLAGS="-O1 -g -s ASSERTIONS=1 -s SAFE_HEAP=1"
EMLINK_FLAGS="-O1 -g -s DEMANGLE_SUPPORT=1"

# 监听文件变化
watch_and_build() {
    echo "监听文件变化..."

    if command -v inotifywait &> /dev/null; then
        # Linux
        inotifywait -mr --format '%w%f' -e modify,create,delete \
            ../../../filament/src/ \
            ../../../web/filament-js/ | while read file; do
            echo "检测到文件变化: $file"
            build_incremental
        done
    elif command -v fswatch &> /dev/null; then
        # macOS
        fswatch -o ../../../filament/src/ ../../../web/filament-js/ | while read; do
            echo "检测到文件变化"
            build_incremental
        done
    else
        echo "警告: 未找到文件监听工具，使用轮询模式"
        while true; do
            sleep 2
            build_incremental
        done
    fi
}

# 增量构建
build_incremental() {
    echo "执行增量构建..."
    start_time=$(date +%s)

    emmake make filament-js 2>&1 | tee build.log

    end_time=$(date +%s)
    duration=$((end_time - start_time))

    if [ ${PIPESTATUS[0]} -eq 0 ]; then
        echo "增量构建成功 (${duration}s)"
        deploy_dev
    else
        echo "增量构建失败 (${duration}s)"
    fi
}

# 部署到开发服务器
deploy_dev() {
    DEV_SERVER_DIR="../../../samples/web"

    if [ -d "$DEV_SERVER_DIR" ]; then
        cp web/filament.js "$DEV_SERVER_DIR/"
        cp web/filament.wasm "$DEV_SERVER_DIR/"
        echo "已部署到开发服务器"
    fi
}
```

### 6.1 资源处理工具

```python
#!/usr/bin/env python3
# tools/asset-processor.py - 资源处理工具

import os
import sys
import gzip
import json
from pathlib import Path

class AssetProcessor:
    def __init__(self, input_dir, output_dir):
        self.input_dir = Path(input_dir)
        self.output_dir = Path(output_dir)
        self.manifest = {}

    def process_all(self):
        """处理所有资源文件"""
        print(f"处理资源: {self.input_dir} -> {self.output_dir}")

        # 创建输出目录
        self.output_dir.mkdir(parents=True, exist_ok=True)

        # 处理不同类型文件
        self.process_gltf_files()
        self.process_texture_files()
        self.process_material_files()

        # 生成资源清单
        self.generate_manifest()

    def process_gltf_files(self):
        """处理 glTF 文件"""
        for gltf_file in self.input_dir.glob("**/*.gltf"):
            print(f"处理 glTF: {gltf_file}")

            # 压缩 glTF 文件
            compressed_file = self.output_dir / f"{gltf_file.stem}.gltf.gz"
            with open(gltf_file, 'rb') as f_in:
                with gzip.open(compressed_file, 'wb') as f_out:
                    f_out.write(f_in.read())

            # 更新清单
            self.manifest[f"models/{gltf_file.name}"] = {
                "compressed": str(compressed_file.relative_to(self.output_dir)),
                "size": compressed_file.stat().st_size,
                "type": "gltf"
            }

    def process_texture_files(self):
        """处理纹理文件"""
        texture_formats = [".jpg", ".png", ".ktx", ".ktx2"]

        for texture_file in self.input_dir.glob("**/*"):
            if texture_file.suffix.lower() in texture_formats:
                print(f"处理纹理: {texture_file}")

                # 生成不同质量版本
                self.generate_texture_variants(texture_file)

    def generate_texture_variants(self, texture_file):
        """生成不同质量的纹理版本"""
        variants = {
            "high": {"quality": 90, "scale": 1.0},
            "medium": {"quality": 75, "scale": 0.75},
            "low": {"quality": 60, "scale": 0.5}
        }

        for variant_name, settings in variants.items():
            output_file = self.output_dir / f"{texture_file.stem}_{variant_name}{texture_file.suffix}"

            # 这里应该调用图像处理库进行缩放和压缩
            # 为简化示例，直接复制文件
            import shutil
            shutil.copy2(texture_file, output_file)

            # 压缩文件
            with open(output_file, 'rb') as f_in:
                with gzip.open(f"{output_file}.gz", 'wb') as f_out:
                    f_out.write(f_in.read())

            # 更新清单
            self.manifest[f"textures/{texture_file.name}"] = {
                "variants": {
                    variant_name: {
                        "file": str(output_file.relative_to(self.output_dir)),
                        "compressed": f"{output_file.relative_to(self.output_dir)}.gz",
                        "size": output_file.stat().st_size
                    }
                }
            }

    def generate_manifest(self):
        """生成资源清单文件"""
        manifest_file = self.output_dir / "manifest.json"

        with open(manifest_file, 'w') as f:
            json.dump(self.manifest, f, indent=2)

        print(f"生成资源清单: {manifest_file}")

if __name__ == "__main__":
    if len(sys.argv) != 3:
        print("用法: python asset-processor.py <输入目录> <输出目录>")
        sys.exit(1)

    processor = AssetProcessor(sys.argv[1], sys.argv[2])
    processor.process_all()
```

### 6.2 性能分析工具

```javascript
// tools/performance-analyzer.js - 性能分析工具

class PerformanceAnalyzer {
    constructor() {
        this.metrics = new Map();
        this.isRecording = false;
        this.startTime = 0;
    }

    startRecording() {
        this.isRecording = true;
        this.startTime = performance.now();
        this.metrics.clear();

        // 监听 WebGL 调用
        this.interceptWebGLCalls();

        // 监听内存使用
        this.monitorMemoryUsage();
    }

    stopRecording() {
        this.isRecording = false;
        return this.generateReport();
    }

    interceptWebGLCalls() {
        const gl = document.querySelector('canvas').getContext('webgl2');
        const originalDrawElements = gl.drawElements.bind(gl);
        const originalDrawArrays = gl.drawArrays.bind(gl);

        let drawCallCount = 0;
        let triangleCount = 0;

        gl.drawElements = function(mode, count, type, offset) {
            if (this.isRecording) {
                drawCallCount++;
                triangleCount += count / 3;
            }
            return originalDrawElements(mode, count, type, offset);
        }.bind(this);

        gl.drawArrays = function(mode, first, count) {
            if (this.isRecording) {
                drawCallCount++;
                triangleCount += count / 3;
            }
            return originalDrawArrays(mode, first, count);
        }.bind(this);

        this.metrics.set('drawCalls', () => drawCallCount);
        this.metrics.set('triangles', () => triangleCount);
    }

    monitorMemoryUsage() {
        const measureMemory = () => {
            if (!this.isRecording) return;

            if (performance.memory) {
                this.metrics.set('jsHeapSize', performance.memory.usedJSHeapSize);
                this.metrics.set('jsHeapLimit', performance.memory.jsHeapSizeLimit);
            }

            if (Module && Module.HEAP8) {
                this.metrics.set('wasmMemory', Module.HEAP8.length);
            }

            setTimeout(measureMemory, 100);
        };

        measureMemory();
    }

    generateReport() {
        const endTime = performance.now();
        const duration = endTime - this.startTime;

        const report = {
            duration: duration,
            timestamp: new Date().toISOString(),
            metrics: {}
        };

        for (const [key, value] of this.metrics.entries()) {
            report.metrics[key] = typeof value === 'function' ? value() : value;
        }

        // 计算性能指标
        report.fps = report.metrics.drawCalls ? (report.metrics.drawCalls / (duration / 1000)) : 0;
        report.memoryEfficiency = report.metrics.jsHeapSize / report.metrics.jsHeapLimit;

        return report;
    }

    exportReport(report, filename = 'performance-report.json') {
        const blob = new Blob([JSON.stringify(report, null, 2)], {
            type: 'application/json'
        });

        const url = URL.createObjectURL(blob);
        const a = document.createElement('a');
        a.href = url;
        a.download = filename;
        a.click();

        URL.revokeObjectURL(url);
    }
}

// 使用示例
const analyzer = new PerformanceAnalyzer();

// 开始录制
analyzer.startRecording();

// 运行渲染循环
setTimeout(() => {
    // 停止录制并生成报告
    const report = analyzer.stopRecording();
    console.log('性能报告:', report);

    // 导出报告
    analyzer.exportReport(report);
}, 10000); // 录制 10 秒
```

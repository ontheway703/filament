# Filament iOS 桥接层与绑定机制

## 桥接层架构

Filament iOS 桥接层通过 Objective-C++ 实现 C++ 引擎与 iOS 平台的无缝集成，提供了平台特定的功能和优化。

```
桥接层架构:
iOS UIKit → Objective-C++ 桥接 → Filament C++ API
    ↓              ↓                    ↓
  UI层           平台适配层             引擎核心
```

#### 初始化流程
```objc
- (instancetype)initWithFrame:(CGRect)frame {
    self = [super initWithFrame:frame];
    if (self) {
        [self setupFilament];
        [self setupDisplayLink];
        [self setupNotifications];
    }
    return self;
}

- (void)setupFilament {
    // 1. 创建引擎
#if FILAMENT_APP_USE_METAL
    engine = Engine::create(Engine::Backend::METAL);
#elif FILAMENT_APP_USE_OPENGL
    engine = Engine::create(Engine::Backend::OPENGL);
#endif

    // 2. 创建渲染器
    renderer = engine->createRenderer();
    
    // 3. 创建场景和视图
    scene = engine->createScene();
    filaView = engine->createView();
    
    // 4. 设置视图和场景的关联
    filaView->setScene(scene);
    
    // 5. 创建相机
    Entity cameraEntity = EntityManager::get().create();
    camera = engine->createCamera(cameraEntity);
    filaView->setCamera(camera);
    
    // 6. 创建交换链
    void* nativeLayer = (__bridge void*)[self getMetalLayer];
    swapChain = engine->createSwapChain(nativeLayer);
    filaView->setViewport({0, 0, (uint32_t)self.bounds.size.width, (uint32_t)self.bounds.size.height});
}
```

### 2. 平台层抽象

**位置**: `filament/backend/src/metal/PlatformMetal.mm`

```objc
class PlatformMetal final : public Platform {
public:
    Driver* createDriver(void* sharedContext, 
                        const Platform::DriverConfig& driverConfig) noexcept override {
        return filament::backend::createMetalDriver(static_cast<Platform*>(this), driverConfig);
    }
    
    void createSwapChain(void* nativeWindow, uint64_t flags) noexcept override {
        CAMetalLayer* metalLayer = (__bridge CAMetalLayer*)nativeWindow;
        
        // 配置 Metal 层属性
        metalLayer.device = MTLCreateSystemDefaultDevice();
        metalLayer.pixelFormat = MTLPixelFormatBGRA8Unorm;
        metalLayer.framebufferOnly = YES;
        
        // 启用 HDR 支持（如果可用）
        if (@available(iOS 16.0, *)) {
            if ([metalLayer.device supportsFamily:MTLGPUFamilyApple7]) {
                metalLayer.wantsExtendedDynamicRangeContent = YES;
            }
        }
    }
    
    uint64_t getOSVersion() const noexcept override {
        NSProcessInfo* processInfo = [NSProcessInfo processInfo];
        NSOperatingSystemVersion version = [processInfo operatingSystemVersion];
        
        // 编码为 64 位整数：主版本(16位) + 次版本(16位) + 补丁(16位)
        return ((uint64_t)version.majorVersion << 32) |
               ((uint64_t)version.minorVersion << 16) |
               (uint64_t)version.patchVersion;
    }
};
```

```objc
    }
```

## 渲染循环集成

### 1. CADisplayLink 集成

```objc
- (void)setupDisplayLink {
    displayLink = [CADisplayLink displayLinkWithTarget:self 
                                              selector:@selector(render)];
    
    // 设置帧率（60fps）
    if (@available(iOS 10.0, *)) {
        displayLink.preferredFramesPerSecond = 60;
    } else {
        displayLink.frameInterval = 1;
    }
    
    [displayLink addToRunLoop:[NSRunLoop mainRunLoop] 
                      forMode:NSRunLoopCommonModes];
}

- (void)render {
    @autoreleasepool {
        // 处理设备旋转
        [self updateDeviceOrientation];
        
        // 执行 Filament 渲染
        if (renderer->beginFrame(swapChain)) {
            renderer->render(filaView);
            renderer->endFrame();
        }
    }
}

- (void)updateDeviceOrientation {
    UIDeviceOrientation orientation = [[UIDevice currentDevice] orientation];
    
    float targetRotation = 0.0f;
    switch (orientation) {
        case UIDeviceOrientationPortrait:
            targetRotation = 0.0f;
            break;
        case UIDeviceOrientationLandscapeLeft:
            targetRotation = M_PI_2;
            break;
        case UIDeviceOrientationLandscapeRight:
            targetRotation = -M_PI_2;
            break;
        case UIDeviceOrientationPortraitUpsideDown:
            targetRotation = M_PI;
            break;
        default:
            return; // 不处理其他方向
    }
    
    // 平滑过渡到目标旋转
    desiredRotation = targetRotation;
    
    const float rotationSpeed = 5.0f;
    float deltaTime = displayLink.duration;
    deviceRotation = lerp(deviceRotation, desiredRotation, rotationSpeed * deltaTime);
    
    // 应用旋转到相机
    math::mat4f rotation = math::mat4f::rotation(deviceRotation, math::float3{0, 0, 1});
    auto& tcm = engine->getTransformManager();
    tcm.setTransform(camera->getEntity(), rotation);
}
```

### 2. 应用状态管理

```objc
- (void)setupNotifications {
    NSNotificationCenter* center = [NSNotificationCenter defaultCenter];
    
    // 应用生命周期通知
    [center addObserver:self 
               selector:@selector(applicationDidBecomeActive:)
                   name:UIApplicationDidBecomeActiveNotification 
                 object:nil];
                 
    [center addObserver:self 
               selector:@selector(applicationWillResignActive:)
                   name:UIApplicationWillResignActiveNotification 
                 object:nil];
                 
    [center addObserver:self 
               selector:@selector(applicationDidEnterBackground:)
                   name:UIApplicationDidEnterBackgroundNotification 
                 object:nil];
    
    // 设备方向通知
    [center addObserver:self 
               selector:@selector(deviceOrientationDidChange:)
                   name:UIDeviceOrientationDidChangeNotification 
                 object:nil];
                 
    // 内存警告通知
    [center addObserver:self 
               selector:@selector(didReceiveMemoryWarning:)
                   name:UIApplicationDidReceiveMemoryWarningNotification 
                 object:nil];
}

- (void)applicationDidBecomeActive:(NSNotification*)notification {
    // 恢复渲染
    displayLink.paused = NO;
}

- (void)applicationWillResignActive:(NSNotification*)notification {
    // 暂停渲染
    displayLink.paused = YES;
}

- (void)applicationDidEnterBackground:(NSNotification*)notification {
    // 完全停止渲染并清理资源
    displayLink.paused = YES;
    
    // 完成当前渲染操作
    engine->flushAndWait();
    
    // 释放非必要资源
    [self cleanupNonEssentialResources];
}

- (void)didReceiveMemoryWarning:(NSNotification*)notification {
    // 清理缓存和非必要资源
    engine->flushAndWait();
    [self cleanupNonEssentialResources];
}
```

## 触摸事件处理

### 1. 手势识别集成

```objc
@interface FilamentView () <UIGestureRecognizerDelegate>
@property (nonatomic, strong) UIPanGestureRecognizer* panGesture;
@property (nonatomic, strong) UIPinchGestureRecognizer* pinchGesture;
@property (nonatomic, strong) UITapGestureRecognizer* tapGesture;
@end

- (void)setupGestureRecognizers {
    // 平移手势（旋转控制）
    self.panGesture = [[UIPanGestureRecognizer alloc] initWithTarget:self 
                                                              action:@selector(handlePan:)];
    self.panGesture.delegate = self;
    [self addGestureRecognizer:self.panGesture];
    
    // 捏合手势（缩放控制）
    self.pinchGesture = [[UIPinchGestureRecognizer alloc] initWithTarget:self 
                                                                  action:@selector(handlePinch:)];
    self.pinchGesture.delegate = self;
    [self addGestureRecognizer:self.pinchGesture];
    
    // 点击手势（选择控制）
    self.tapGesture = [[UITapGestureRecognizer alloc] initWithTarget:self 
                                                              action:@selector(handleTap:)];
    self.tapGesture.delegate = self;
    [self addGestureRecognizer:self.tapGesture];
}

- (void)handlePan:(UIPanGestureRecognizer*)gesture {
    CGPoint translation = [gesture translationInView:self];
    
    // 转换为相机旋转
    const float sensitivity = 0.01f;
    app.cameraController.rotationX += translation.y * sensitivity;
    app.cameraController.rotationY += translation.x * sensitivity;
    
    // 限制垂直旋转范围
    app.cameraController.rotationX = fmaxf(-M_PI_2 + 0.1f, 
                                         fminf(M_PI_2 - 0.1f, app.cameraController.rotationX));
    
    [self updateCameraTransform];
    [gesture setTranslation:CGPointZero inView:self];
}

- (void)handlePinch:(UIPinchGestureRecognizer*)gesture {
    static float initialDistance = 0.0f;
    
    if (gesture.state == UIGestureRecognizerStateBegan) {
        initialDistance = app.cameraController.distance;
    }
    
    // 缩放相机距离
    float scale = gesture.scale;
    app.cameraController.distance = initialDistance / scale;
    
    // 限制缩放范围
    app.cameraController.distance = fmaxf(1.0f, fminf(100.0f, app.cameraController.distance));
    
    [self updateCameraTransform];
}

- (void)handleTap:(UITapGestureRecognizer*)gesture {
    CGPoint location = [gesture locationInView:self];
    
    // 转换为归一化坐标
    float x = (location.x / self.bounds.size.width) * 2.0f - 1.0f;
    float y = -((location.y / self.bounds.size.height) * 2.0f - 1.0f);
    
    // 执行射线拾取
    [self performRayPickingAtX:x y:y];
}
```

### 2. 相机控制器

```objc
struct CameraController {
    float rotationX = 0.0f;     // 垂直旋转
    float rotationY = 0.0f;     // 水平旋转
    float distance = 5.0f;      // 距离目标的距离
    math::float3 target = {0, 0, 0};  // 观察目标
};

- (void)updateCameraTransform {
    // 计算相机位置
    math::float3 position;
    position.x = app.cameraController.target.x + 
                app.cameraController.distance * 
                cosf(app.cameraController.rotationX) * 
                sinf(app.cameraController.rotationY);
                
    position.y = app.cameraController.target.y + 
                app.cameraController.distance * 
                sinf(app.cameraController.rotationX);
                
    position.z = app.cameraController.target.z + 
                app.cameraController.distance * 
                cosf(app.cameraController.rotationX) * 
                cosf(app.cameraController.rotationY);
    
    // 设置相机观察变换
    camera->lookAt(position, app.cameraController.target, {0, 1, 0});
}
```

## 资源管理桥接

### 1. 纹理加载桥接

```objc
- (Texture*)loadTextureFromBundle:(NSString*)name {
    // 从应用包加载图像
    NSString* path = [[NSBundle mainBundle] pathForResource:name ofType:nil];
    UIImage* image = [UIImage imageWithContentsOfFile:path];
    
    if (!image) {
        utils::slog.e << "Failed to load image: " << [name UTF8String];
        return nullptr;
    }
    
    // 转换为像素数据
    CGImageRef cgImage = image.CGImage;
    size_t width = CGImageGetWidth(cgImage);
    size_t height = CGImageGetHeight(cgImage);
    
    // 创建像素缓冲区
    std::vector<uint8_t> pixelData(width * height * 4);
    
    CGColorSpaceRef colorSpace = CGColorSpaceCreateDeviceRGB();
    CGContextRef context = CGBitmapContextCreate(
        pixelData.data(),
        width, height,
        8, width * 4,
        colorSpace,
        kCGImageAlphaPremultipliedLast | kCGBitmapByteOrder32Big
    );
    
    CGContextDrawImage(context, CGRectMake(0, 0, width, height), cgImage);
    
    CGContextRelease(context);
    CGColorSpaceRelease(colorSpace);
    
    // 创建 Filament 纹理
    Texture* texture = Texture::Builder()
        .width(static_cast<uint32_t>(width))
        .height(static_cast<uint32_t>(height))
        .levels(1)
        .format(Texture::InternalFormat::RGBA8)
        .build(*engine);
    
    // 上传像素数据
    Texture::PixelBufferDescriptor buffer(
        pixelData.data(),
        pixelData.size(),
        Texture::Format::RGBA,
        Texture::Type::UBYTE
    );
    
    texture->setImage(*engine, 0, std::move(buffer));
    
    return texture;
}
```

```objc
        }
    }
}
```

### 3. 网络资源加载

```objc
- (void)loadModelFromURL:(NSURL*)url completion:(void(^)(gltfio::FilamentAsset*))completion {
    NSURLSessionDataTask* task = [[NSURLSession sharedSession] 
        dataTaskWithURL:url 
        completionHandler:^(NSData* data, NSURLResponse* response, NSError* error) {
            
        if (error || !data) {
            utils::slog.e << "Failed to download model: " << [error.localizedDescription UTF8String];
            dispatch_async(dispatch_get_main_queue(), ^{
                completion(nullptr);
            });
            return;
        }
        
        // 在后台线程解析模型
        dispatch_async(dispatch_get_global_queue(DISPATCH_QUEUE_PRIORITY_DEFAULT, 0), ^{
            gltfio::AssetLoader* loader = app.assetLoader;
            gltfio::FilamentAsset* asset = loader->createAsset(
                (const uint8_t*)data.bytes, data.length
            );
            
            dispatch_async(dispatch_get_main_queue(), ^{
                completion(asset);
            });
        });
    }];
    
    [task resume];
}
```

## 内存管理和生命周期

### 1. 自动引用计数集成

```objc
- (void)dealloc {
    // 1. 停止渲染循环
    [displayLink invalidate];
    displayLink = nil;
    
    // 2. 移除通知监听
    [[NSNotificationCenter defaultCenter] removeObserver:self];
    
    // 3. 清理 Filament 资源（按依赖顺序）
    if (engine) {
        // 先销毁依赖对象
        if (renderer) {
            engine->destroy(renderer);
            renderer = nullptr;
        }
        
        if (scene) {
            engine->destroy(scene);
            scene = nullptr;
        }
        
        if (filaView) {
            engine->destroy(filaView);
            filaView = nullptr;
        }
        
        if (camera) {
            engine->destroy(camera);
            camera = nullptr;
        }
        
        if (swapChain) {
            engine->destroy(swapChain);
            swapChain = nullptr;
        }
        
        // 最后销毁引擎
        Engine::destroy(engine);
        engine = nullptr;
    }
}
```

### 2. 异常处理

```objc
- (BOOL)safeExecuteFilamentOperation:(void(^)(void))operation {
    @try {
        operation();
        return YES;
    }
    @catch (NSException* exception) {
        utils::slog.e << "Filament operation failed: " 
                     << [exception.reason UTF8String];
        return NO;
    }
}

- (void)handleFilamentError:(const std::exception&)error {
    NSString* errorMessage = [NSString stringWithUTF8String:error.what()];
    utils::slog.e << "Filament C++ exception: " << [errorMessage UTF8String];
    
    // 可选：显示错误对话框
    UIAlertController* alert = [UIAlertController 
        alertControllerWithTitle:@"Rendering Error"
                         message:errorMessage
                  preferredStyle:UIAlertControllerStyleAlert];
                  
    [alert addAction:[UIAlertAction actionWithTitle:@"OK" 
                                              style:UIAlertActionStyleDefault 
                                            handler:nil]];
                                            
    [[UIApplication sharedApplication].keyWindow.rootViewController 
        presentViewController:alert animated:YES completion:nil];
}
```

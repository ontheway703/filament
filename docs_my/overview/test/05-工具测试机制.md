# Filament 工具测试机制

## 工具测试概述

Filament 提供了一系列构建工具和辅助工具，每个工具都有专门的测试来验证其功能正确性。这些测试确保工具在不同平台和使用场景下的稳定性。

## 工具分类

### 1. 材质相关工具
- **matc** - 材质编译器
- **filamat** - 材质处理库

### 2. 资产处理工具
- **cmgen** - 环境贴图生成器
- **filamesh** - 网格转换器
- **mipgen** - Mipmap 生成器

### 3. 着色器工具
- **glslminifier** - GLSL 着色器压缩器

### 4. 图像处理工具
- **image** - 图像格式转换库
- **ktxreader** - KTX 纹理读取器

## 环境贴图生成器测试 (cmgen)

### 测试结构

**位置**: `tools/cmgen/tests/`
**主要测试**: `test_cmgen.cpp`

### 功能测试

#### 基础图像处理
```cpp
// test_cmgen.cpp
#include <gtest/gtest.h>
#include <cmgen/CubemapUtils.h>
#include <image/LinearImage.h>

using namespace image;
using namespace cmgen;

class CmgenTest : public ::testing::Test {
protected:
    void SetUp() override {
        // 创建测试用的环境贴图
        testCubemap = createTestCubemap(256, 256);
    }

    std::array<LinearImage, 6> testCubemap;

    LinearImage createTestCubemap(size_t width, size_t height) {
        // 生成渐变测试图像
        LinearImage image(width, height, 3);
        float3* pixels = reinterpret_cast<float3*>(image.getPixelRef());

        for (size_t y = 0; y < height; ++y) {
            for (size_t x = 0; x < width; ++x) {
                float u = (x + 0.5f) / width;
                float v = (y + 0.5f) / height;
                pixels[y * width + x] = float3(u, v, 0.5f);
            }
        }

        return image;
    }
};

TEST_F(CmgenTest, GenerateIrradianceMap) {
    CubemapUtils utils;

    // 生成辐照度贴图
    auto irradiance = utils.irradiance(testCubemap, 64);

    // 验证输出
    EXPECT_EQ(irradiance[0].getWidth(), 64);
    EXPECT_EQ(irradiance[0].getHeight(), 64);

    // 验证能量守恒
    float totalEnergy = computeTotalEnergy(irradiance);
    EXPECT_GT(totalEnergy, 0.0f);
    EXPECT_LT(totalEnergy, 1000.0f); // 合理范围
}

TEST_F(CmgenTest, GeneratePrefilterMap) {
    CubemapUtils utils;

    // 生成预过滤贴图（多级mipmap）
    for (int level = 0; level < 5; ++level) {
        float roughness = level / 4.0f;
        auto prefiltered = utils.prefilter(testCubemap,
                                         128 >> level, roughness);

        EXPECT_EQ(prefiltered[0].getWidth(), 128 >> level);

        // 验证粗糙度增加时图像变模糊
        if (level > 0) {
            float sharpness = computeSharpness(prefiltered);
            EXPECT_LT(sharpness, previousSharpness);
        }
    }
}
```

```cpp
// test_filamesh.cpp
#include <gtest/gtest.h>
#include <filameshio/MeshReader.h>

using namespace filameshio;

class FilameshTest : public ::testing::Test {
protected:
    void SetUp() override {
        // 创建简单的三角形 OBJ 数据
        objData = R"(
            v 0.0 0.0 0.0
            v 1.0 0.0 0.0
            v 0.5 1.0 0.0
            vn 0.0 0.0 1.0
            vt 0.0 0.0
            vt 1.0 0.0
            vt 0.5 1.0
            f 1/1/1 2/2/1 3/3/1
        )";
    }

    std::string objData;
};

TEST_F(FilameshTest, LoadOBJTriangle) {
    MeshReader reader;

    Mesh mesh;
    bool success = reader.loadFromMemory(objData.data(),
                                       objData.size(), mesh);

    EXPECT_TRUE(success);
    EXPECT_EQ(mesh.vertexCount, 3);
    EXPECT_EQ(mesh.triangleCount, 1);

    // 验证顶点数据
    EXPECT_FLOAT_EQ(mesh.positions[0], 0.0f);
    EXPECT_FLOAT_EQ(mesh.positions[1], 0.0f);
    EXPECT_FLOAT_EQ(mesh.positions[2], 0.0f);

    // 验证法线
    EXPECT_FLOAT_EQ(mesh.normals[2], 1.0f);

    // 验证纹理坐标
    EXPECT_FLOAT_EQ(mesh.texCoords[0], 0.0f);
    EXPECT_FLOAT_EQ(mesh.texCoords[1], 0.0f);
}
```

#### Filamesh 二进制格式
```cpp
TEST_F(FilameshTest, SaveAndLoadFilameshFormat) {
    MeshReader reader;
    MeshWriter writer;

    // 加载 OBJ
    Mesh originalMesh;
    reader.loadFromMemory(objData.data(), objData.size(), originalMesh);

    // 保存为 Filamesh 格式
    std::vector<uint8_t> filameshData;
    writer.writeToMemory(originalMesh, filameshData);

    // 重新加载
    Mesh loadedMesh;
    bool success = reader.loadFilamesh(filameshData.data(),
                                     filameshData.size(), loadedMesh);

    EXPECT_TRUE(success);

    // 验证数据一致性
    EXPECT_EQ(loadedMesh.vertexCount, originalMesh.vertexCount);
    EXPECT_EQ(loadedMesh.triangleCount, originalMesh.triangleCount);

    // 逐顶点比较
    for (size_t i = 0; i < originalMesh.vertexCount * 3; ++i) {
        EXPECT_FLOAT_EQ(loadedMesh.positions[i], originalMesh.positions[i]);
    }
}
```

#### 基础压缩
```cpp
// test_glslminifier.cpp
#include <gtest/gtest.h>
#include <glslminifier/GlslMinifier.h>

class GlslMinifierTest : public ::testing::Test {
protected:
    std::string testShader = R"(
        #version 450 core

        // 顶点着色器输入
        layout(location = 0) in vec3 position;
        layout(location = 1) in vec3 normal;
        layout(location = 2) in vec2 texCoord;

        // 输出到片段着色器
        out vec3 worldNormal;
        out vec2 uv;

        // Uniform 变量
        uniform mat4 modelViewProjection;
        uniform mat3 normalMatrix;

        void main() {
            // 计算世界空间法线
            worldNormal = normalize(normalMatrix * normal);

            // 传递纹理坐标
            uv = texCoord;

            // 计算最终位置
            gl_Position = modelViewProjection * vec4(position, 1.0);
        }
    )";
};

TEST_F(GlslMinifierTest, BasicMinification) {
    GlslMinifier minifier;

    std::string minified;
    bool success = minifier.minify(testShader, minified);

    EXPECT_TRUE(success);
    EXPECT_LT(minified.length(), testShader.length());

    // 验证关键元素仍然存在
    EXPECT_NE(minified.find("main"), std::string::npos);
    EXPECT_NE(minified.find("gl_Position"), std::string::npos);

    // 验证注释被移除
    EXPECT_EQ(minified.find("//"), std::string::npos);
    EXPECT_EQ(minified.find("/*"), std::string::npos);
}

TEST_F(GlslMinifierTest, PreserveSemantics) {
    GlslMinifier minifier;
    minifier.setPreservePrecision(true);

    std::string minified;
    bool success = minifier.minify(testShader, minified);

    EXPECT_TRUE(success);

    // 编译验证（需要 GLSL 编译器）
    bool compileResult = validateGLSL(minified);
    EXPECT_TRUE(compileResult);
}
```

```cpp
TEST_F(ImageTest, sRGBToLinearConversion) {
    LinearImage srgbImage = createTestImage(64, 64);

    // 应用 sRGB 到线性变换
    auto linearImage = ColorTransform::sRGBToLinear(srgbImage);

    // 验证转换结果
    const float* srgbPixels = srgbImage.getPixelRef();
    const float* linearPixels = linearImage.getPixelRef();

        }
    }
}
```

```cpp
#include <gtest/gtest.h>
#include <ktxreader/Ktx2Reader.h>

class KtxReaderTest : public ::testing::Test {
protected:
    void SetUp() override {
        // 加载测试 KTX 文件
        testFile = loadTestFile("color_grid_uastc_zstd.ktx2");
    }

    std::vector<uint8_t> testFile;
};

TEST_F(KtxReaderTest, LoadKTX2File) {
    Ktx2Reader reader(testFile);

    EXPECT_TRUE(reader.isValid());
    EXPECT_GT(reader.info().pixelWidth, 0);
    EXPECT_GT(reader.info().pixelHeight, 0);

    // 验证格式信息
    auto info = reader.info();
    EXPECT_EQ(info.vkFormat, VK_FORMAT_ASTC_4x4_UNORM_BLOCK);
    EXPECT_TRUE(info.isCompressed);

TEST_F(KtxReaderTest, DecompressTexture) {
    Ktx2Reader reader(testFile);

    std::vector<uint8_t> decompressed;
    bool success = reader.transcode(ktx2::TranscodeFormat::RGBA8,
                                  decompressed);

    EXPECT_TRUE(success);
    EXPECT_FALSE(decompressed.empty());

    // 验证解压数据大小
    auto info = reader.info();
    size_t expectedSize = info.pixelWidth * info.pixelHeight * 4;
    EXPECT_EQ(decompressed.size(), expectedSize);
}
```

## 测试执行集成

### 构建配置
所有工具测试都通过 CMake 集成到主构建系统：

```cmake
# 条件编译

    # 桌面平台才构建工具测试
    add_subdirectory(tools/matc/tests)
    add_subdirectory(tools/cmgen/tests)
    # ...
endif()
```

### 测试列表集成
工具测试包含在 `build/common/test_list.txt` 中：

```

libs/image/test_image compare libs/image/tests/reference/
```

### 参考数据管理
某些测试需要参考数据进行对比：

- **image 测试**: `libs/image/tests/reference/`

这些工具测试确保了 Filament 工具链的可靠性和正确性。

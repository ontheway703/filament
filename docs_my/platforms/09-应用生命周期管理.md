# 应用生命周期管理

## 概述

正确处理应用生命周期事件对于 Filament 应用的稳定性和性能至关重要。本文档详细介绍各平台的生命周期管理、资源清理和状态恢复策略。

**主要内容**:
- ✅ 各平台生命周期事件
- ✅ 资源管理和清理
- ✅ 状态保存和恢复
- ✅ 后台/前台切换处理
- ✅ 内存警告响应
- ✅ 最佳实践和常见陷阱

## Android 生命周期

### Activity 生命周期

```java
public class FilamentActivity extends Activity {
    private long nativeEngine;
    private long nativeSwapChain;
    private long nativeRenderer;
    private long nativeView;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);

        // 创建 Filament 引擎
        nativeEngine = nativeCreateEngine();
    }

    @Override
    protected void onStart() {
        super.onStart();
        // Activity 可见
    }

    @Override
    protected void onResume() {
        super.onResume();

        // 恢复渲染循环
        Choreographer.getInstance().postFrameCallback(frameCallback);
    }

    @Override
    protected void onPause() {
        super.onPause();

        // 暂停渲染循环
        Choreographer.getInstance().removeFrameCallback(frameCallback);

        // 等待当前帧完成
        nativeFlushAndWait(nativeEngine);
    }

    @Override
    protected void onStop() {
        super.onStop();
        // Activity 不可见
    }

    @Override
    protected void onDestroy() {
        super.onDestroy();

        // 清理资源
        nativeDestroyView(nativeView);
        nativeDestroyRenderer(nativeRenderer);
        nativeDestroySwapChain(nativeSwapChain);
        nativeDestroyEngine(nativeEngine);
    }
}
```

### 内存警告处理

```java
@Override
public void onTrimMemory(int level) {
    super.onTrimMemory(level);

    switch (level) {
        case TRIM_MEMORY_RUNNING_CRITICAL:
        case TRIM_MEMORY_COMPLETE:
            // 释放所有可释放的资源
            nativeReduceMemory(nativeEngine, MEMORY_LEVEL_CRITICAL);
            break;

        case TRIM_MEMORY_RUNNING_LOW:
        case TRIM_MEMORY_MODERATE:
            // 释放部分资源
            nativeReduceMemory(nativeEngine, MEMORY_LEVEL_LOW);
            break;

        case TRIM_MEMORY_RUNNING_MODERATE:
            // 释放缓存
            nativeReduceMemory(nativeEngine, MEMORY_LEVEL_MODERATE);
            break;
    }
}
```

```objc
    [[NSNotificationCenter defaultCenter] addObserver:self
                                             selector:@selector(appDidEnterBackground:)
                                                 name:UIApplicationDidEnterBackgroundNotification
                                               object:nil];

    [[NSNotificationCenter defaultCenter] addObserver:self
                                             selector:@selector(appWillEnterForeground:)
                                                 name:UIApplicationWillEnterForegroundNotification
                                               object:nil];

    [[NSNotificationCenter defaultCenter] addObserver:self
                                             selector:@selector(appWillTerminate:)
                                                 name:UIApplicationWillTerminateNotification
                                               object:nil];
}

- (void)appDidEnterBackground:(NSNotification*)notification {
    // 应用进入后台
    [self stopDisplayLink];

    // 等待渲染完成
    [self.filamentRenderer flushAndWait];

    // 释放可释放的资源
    [self.filamentRenderer reduceMemoryUsage];
}

- (void)appWillEnterForeground:(NSNotification*)notification {
    // 应用回到前台
    [self startDisplayLink];

    // 恢复资源
    [self.filamentRenderer restoreResources];
}

- (void)appWillTerminate:(NSNotification*)notification {
    // 应用即将终止
    [self.filamentRenderer cleanup];
}

- (void)dealloc {
    [[NSNotificationCenter defaultCenter] removeObserver:self];
}
```

## Windows 生命周期

### 窗口消息处理

```cpp
LRESULT CALLBACK WndProc(HWND hwnd, UINT msg, WPARAM wParam, LPARAM lParam) {
    auto* app = reinterpret_cast<FilamentApp*>(GetWindowLongPtr(hwnd, GWLP_USERDATA));

    switch (msg) {
    case WM_CREATE:
        // 窗口创建
        return 0;

    case WM_SIZE: {
        // 窗口大小变化
        UINT width = LOWORD(lParam);
        UINT height = HIWORD(lParam);

        if (wParam == SIZE_MINIMIZED) {
            // 窗口最小化 - 暂停渲染
            if (app) app->pause();
        } else if (wParam == SIZE_MAXIMIZED || wParam == SIZE_RESTORED) {
            // 窗口最大化或恢复 - 恢复渲染
            if (app) {
                app->resume();
                app->resize(width, height);
            }
        }
        return 0;
    }

    case WM_ACTIVATE:
        if (LOWORD(wParam) == WA_INACTIVE) {
            // 窗口失去焦点
            if (app) app->pause();
        } else {
            // 窗口获得焦点
            if (app) app->resume();
        }
        return 0;

    case WM_CLOSE:
        // 窗口关闭
        DestroyWindow(hwnd);
        return 0;

    case WM_DESTROY:
        // 清理资源
        if (app) {
            app->cleanup();
        }
        PostQuitMessage(0);
        return 0;

    default:
        return DefWindowProc(hwnd, msg, wParam, lParam);
    }
}
```

### 应用状态管理

```cpp
class FilamentApp {
    bool m_isActive = true;
    bool m_isMinimized = false;

public:
    void pause() {
        m_isActive = false;

        // 等待渲染完成
        if (m_engine) {
            m_engine->flushAndWait();
        }
    }

    void resume() {
        m_isActive = true;
    }

    void render() {
        if (!m_isActive || m_isMinimized) {
            return;  // 跳过渲染
        }

        if (m_renderer->beginFrame(m_swapChain)) {
            m_renderer->render(m_view);
            m_renderer->endFrame();
        }
    }
};
```

## Linux 生命周期

### X11 事件处理

```cpp
void handleX11Events(Display* display, FilamentApp* app) {
    XEvent event;

    while (XPending(display)) {
        XNextEvent(display, &event);

        switch (event.type) {
        case ConfigureNotify:
            // 窗口大小变化
            if (event.xconfigure.width != app->width ||
                event.xconfigure.height != app->height) {
                app->resize(event.xconfigure.width, event.xconfigure.height);
            }
            break;

        case UnmapNotify:
            // 窗口被隐藏
            app->pause();
            break;

        case MapNotify:
            // 窗口被显示
            app->resume();
            break;

        case DestroyNotify:
            // 窗口销毁
            app->cleanup();
            break;

        case ClientMessage:
            // 窗口管理器消息 (如关闭窗口)
            if (event.xclient.data.l[0] == app->wmDeleteMessage) {
                app->cleanup();
                XCloseDisplay(display);
                exit(0);
            }
            break;
        }
    }
}
```

### 信号处理

```cpp
#include <signal.h>

volatile sig_atomic_t g_shouldExit = 0;

void signalHandler(int signal) {
    g_shouldExit = 1;
}

int main() {
    // 注册信号处理器
    signal(SIGINT, signalHandler);   // Ctrl+C
    signal(SIGTERM, signalHandler);  // kill
    signal(SIGHUP, signalHandler);   // 终端关闭

    FilamentApp app;

    while (!g_shouldExit) {
        app.render();
    }

    app.cleanup();
    return 0;
}
```

### 资源清理

```javascript
function cleanup() {
    if (engine) {
        // 销毁资源
        engine.destroyView(view);
        engine.destroyScene(scene);
        engine.destroyRenderer(renderer);
        engine.destroySwapChain(swapChain);

        // 销毁所有实体
        const entities = scene.getEntities();
        for (let i = 0; i < entities.length; i++) {
            engine.destroyEntity(entities[i]);
        }

        // 销毁引擎
        Filament.Engine.destroy(engine);
        engine = null;
    }
}
```

## 资源管理策略

### RAII 模式 (C++)

```cpp
class FilamentContext {
public:
    FilamentContext(void* nativeWindow) {
        m_engine = Engine::create();
        m_swapChain = m_engine->createSwapChain(nativeWindow);
        m_renderer = m_engine->createRenderer();
        m_scene = m_engine->createScene();
        m_view = m_engine->createView();
    }

    ~FilamentContext() {
        cleanup();
    }

    // 禁用拷贝
    FilamentContext(const FilamentContext&) = delete;
    FilamentContext& operator=(const FilamentContext&) = delete;

    // 支持移动
    FilamentContext(FilamentContext&& other) noexcept {
        *this = std::move(other);
    }

    FilamentContext& operator=(FilamentContext&& other) noexcept {
        if (this != &other) {
            cleanup();
            m_engine = other.m_engine;
            m_swapChain = other.m_swapChain;
            // ... 其他成员
            other.m_engine = nullptr;
            other.m_swapChain = nullptr;
        }
        return *this;
    }

private:
    void cleanup() {
        if (m_engine) {
            if (m_view) m_engine->destroy(m_view);
            if (m_scene) m_engine->destroy(m_scene);
            if (m_renderer) m_engine->destroy(m_renderer);
            if (m_swapChain) m_engine->destroy(m_swapChain);
            Engine::destroy(&m_engine);
        }
    }

    Engine* m_engine = nullptr;
    SwapChain* m_swapChain = nullptr;
    Renderer* m_renderer = nullptr;
    Scene* m_scene = nullptr;
    View* m_view = nullptr;
};
```

```objc
        _engine = nullptr;

    self.isInitialized = NO;
}

- (void)dealloc {
    [self cleanup];
}

@end
```

```objc
- (void)encodeRestorableStateWithCoder:(NSCoder*)coder {
    [super encodeRestorableStateWithCoder:coder];

    // 保存相机状态
    simd_float3 cameraPos = [self.filamentRenderer getCameraPosition];
    [coder encodeBytes:(const uint8_t*)&cameraPos length:sizeof(cameraPos) forKey:@"cameraPosition"];

    [super decodeRestorableStateWithCoder:coder];

    // 恢复相机状态
    NSUInteger length;
    const uint8_t* bytes = [coder decodeBytesForKey:@"cameraPosition" returnedLength:&length];
    if (bytes && length == sizeof(simd_float3)) {
        simd_float3 cameraPos;
        memcpy(&cameraPos, bytes, sizeof(cameraPos));
        [self.filamentRenderer setCameraPosition:cameraPos];
    }
}
```

## 最佳实践

### 1. 延迟资源清理

```cpp
// 坏例子: 立即销毁可能导致崩溃
void onDestroy() {
    engine->destroy(swapChain);  // 可能还在使用中
}

// 好例子: 等待渲染完成
void onDestroy() {
    engine->flushAndWait();  // 等待所有渲染完成
    engine->destroy(swapChain);
}
```

### 2. 避免重复初始化

```cpp
class FilamentApp {
    bool m_initialized = false;

public:
    void onCreate() {
        if (m_initialized) return;

        m_engine = Engine::create();
        m_initialized = true;
    }

    void onDestroy() {
        if (!m_initialized) return;

        cleanup();
        m_initialized = false;
    }
};
```

### 3. 渐进式资源加载

```cpp
// 在后台加载资源
void loadResources() {
    // 1. 加载必需资源 (阻塞)
    loadCriticalAssets();

    // 2. 异步加载其他资源
    std::thread([this]() {
        loadOptionalAssets();
    }).detach();
}
```

### 4. 内存压力响应

```cpp
void onLowMemory() {
    // 1. 清空缓存
    textureCache.clear();

    // 2. 降低 LOD
    for (auto& model : models) {
        model.setLOD(LOD_LOW);
```

## 常见陷阱

### 1. 忘记停止渲染循环

```cpp
// 坏例子
~FilamentApp() {
    cleanup();  // 渲染循环可能还在运行!
}

// 好例子
~FilamentApp() {
    stopRenderLoop();  // 先停止循环
    cleanup();         // 再清理资源
}
```

### 2. 不等待渲染完成

```cpp
// 坏例子
void onPause() {
    // 直接清理,可能导致崩溃
    engine->destroy(swapChain);
}

// 好例子
void onPause() {
    engine->flushAndWait();  // 等待完成
    engine->destroy(swapChain);
}
```

### 3. 泄漏资源

```cpp
// 坏例子
void onCreate() {
    m_engine = Engine::create();
    m_swapChain = m_engine->createSwapChain(window);
}

void onCreate() {  // 再次调用!
    m_engine = Engine::create();  // 泄漏了之前的 engine!
}

// 好例子
void onCreate() {
    if (m_engine) {
        cleanup();  // 先清理旧资源
    }
    m_engine = Engine::create();
}
```
